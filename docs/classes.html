Stack (LIFO)

Создание: конструктор Stack() или литерал #stack{}

var s = Stack();
s.push(10);
s.push(20);
s.push(30);
print(s.top());    // 30
print(s.pop());    // 30
print(s.peek());   // 20
print(s.size);     // 2
Свойство/МетодОписание
s.sizeКоличество элементов (свойство)
s.emptytrue если пуст (свойство)
s[i]Доступ по индексу (0 = верх)
s.push(val)Добавление наверх
s.top()Возврат верхнего без удаления
s.peek()Алиас top()
s.pop()Удаление и возврат верхнего
s.clear()Очистка
s.reserve(n)Резервирование памяти

Queue (FIFO)

Создание: конструктор Queue() или литерал #queue{}

var q = Queue();
q.push(10);
q.push(20);
q.push(30);
print(q.front());  // 10
print(q.pop());    // 10
print(q.back());   // 30
МетодОписание
q.push(val)Добавление в конец
q.push_front(val)Добавление в начало
q.pop()Удаление первого
q.pop_back()Удаление последнего
q.front()Первый элемент
q.back()Последний элемент
q[i]Доступ по индексу
q.sizeКоличество элементов
q.emptytrue если пуст

PriorityQueue (max-heap)

Создание: конструктор PriorityQueue() или литерал #priorityqueue{}

Чем больше число приоритета — тем выше приоритет (pop извлекает максимальный).

var pq = PriorityQueue();
pq.push("low", 1);
pq.push("high", 10);
pq.push("medium", 5);
print(pq.peek());  // "high"
print(pq.pop());   // "high" (приоритет 10)
print(pq.pop());   // "medium" (приоритет 5)
МетодОписание
pq.push(val, priority)Добавление с приоритетом (по умолчанию 0)
pq.pop()Извлечение с наивысшим приоритетом
pq.peek()Просмотр без удаления
pq.sizeКоличество элементов
pq.emptytrue если пуст
pq.clear()Очистка

Dictionary (case-sensitive)

Создание: конструктор Dictionary() или литерал #dictionary{}

var d = Dictionary();
d.set("name", "Иван");
d.set("age", 25);
print(d.get("name"));    // "Иван"
print(d.has("age"));     // true
print(d.size);            // 2

// Доступ через точку
print(d.name);            // "Иван"

// Проверка ключа
print(d.contains("name"));  // true
МетодОписание
d.set(key, val)Установка значения
d.get(key)Получение значения
d.getdefault(key, def)С значением по умолчанию
d.has(key)Наличие ключа
d.contains(key)Алиас has()
d.remove(key)Удаление
d.pop(key)Удаление и возврат
d.keys()Массив ключей
d.values()Массив значений
d.items()Массив кортежей [key, value]
d.update(other)Обновление из другого словаря
d.clear()Очистка

DictionaryCI (case-insensitive)

Создание: конструктор DictionaryCI() или литерал #dictionaryci{}

Аналог Dictionary, но ключи без учёта регистра.

var d = DictionaryCI();
d.set("Name", "Иван");
print(d.get("name"));  // "Иван" — регистр не важен

Mutex

var m = Mutex();
m.lock();
// критическая секция
m.unlock();
Свойство/МетодОписание
m.lockedtrue если заблокирован
m.lock()Блокировка
m.tryLock()Попытка блокировки (true/false)
m.unlock()Разблокировка

LockGuard (RAII)

Автоматическая разблокировка при выходе из scope.

var m = Mutex();
{
    var guard = LockGuard(m);
    // критическая секция
    guard.dismiss();  // необязательно: отпустить раньше
}  // автоматическая разблокировка

synchronized()

var m = Mutex();
synchronized(m, function() {
    // атомарный блок
    sharedData += 1;
});
