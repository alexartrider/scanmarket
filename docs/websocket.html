Плагин WebSocket

WebSocket клиент — двусторонняя связь в реальном времени, текстовые и бинарные сообщения, ping/pong

RUEN

Главная / Плагины / WebSocket

Зависимости

Требуются OpenSSL 3: libssl-3-x64.dll, libcrypto-3-x64.dll (размещаются рядом с бинарником).

Входящие текстовые и бинарные сообщения собираются из WebSocket-фрагментов; общий размер сообщения ограничен 64 MiB.

Загрузка плагина

loadPlugin("plugins/websocket.dll");

Создание объекта

var ws = WebSocket();

Подключение

Поддерживается ws:// (обычный) и wss:// (SSL/TLS).

var ws = WebSocket();

// ws://
ws.connect("ws://echo.websocket.events");

// wss://
ws.connect("wss://echo.websocket.events");

// с портом
ws.connect("ws://localhost:8080");

// с путём
ws.connect("wss://example.com/chat");

print(ws.isconnected());  // 1

Текстовые сообщения

var ws = WebSocket();
ws.connect("wss://echo.websocket.events");

// Отправка
ws.send("Привет, мир!");
ws.send("val=\(42)");

// Получение
var msg = ws.recv();
if (msg != null) {
    print("Получено: \(msg)");
}

// Проверка наличия сообщений
if (ws.hasMessage()) {
    var next = ws.recv();
}

Бинарные сообщения

var ws = WebSocket();
ws.connect("wss://echo.websocket.events");

// Отправка бинарных данных (ByteArray)
var ba = ByteArray();
ba.push(0x48);  // H
ba.push(0x65);  // e
ba.push(0x6C);  // l
ba.push(0x6C);  // l
ba.push(0x6F);  // o
ws.sendBinary(ba);

// Получение бинарных данных
var data = ws.recvBinary();
if (data != null) {
    print("Размер: \(data.size())");
    print("Первый байт: \(data[0])");
}

// Отправка строки как бинарных данных
ws.sendBinary("raw bytes");

Ping / Pong

Плагин автоматически отвечает на ping-фреймы сервера pong-фреймами.

var ws = WebSocket();
ws.connect("wss://echo.websocket.events");

// Ручная отправка ping
ws.ping();

Закрытие соединения

var ws = WebSocket();
ws.connect("wss://echo.websocket.events");

// Корректное закрытие (отправляет close-фрейм)
ws.close();

// Некорректное отключение (без close-фрейма)
ws.disconnect();

Очередь сообщений

var ws = WebSocket();
ws.connect("wss://echo.websocket.events");

// Количество сообщений в очереди
print(ws.messageCount());

// Есть ли сообщения
print(ws.hasMessage());  // 1 или 0

// Обработка всех сообщений
while (ws.hasMessage()) {
    var msg = ws.recv();
    print(msg);
}

// Очистка очереди
ws.clearMessages();

Таймаут

var ws = WebSocket();
ws.connect("wss://echo.websocket.events");

// Установка таймаута чтения (мсек)
ws.setTimeout(5000);  // 5 секунд

// Получение текущего таймаута
var timeout = ws.setTimeout();  // геттер

Пример: Echo-клиент

loadPlugin("plugins/websocket.dll");

var ws = WebSocket();

if (ws.connect("wss://echo.websocket.events")) {
    print("Подключено!");

    // Отправка сообщений
    ws.send("Привет!");
    ws.send("Echo test");

    // Ожидание ответов
    system.sleep(2000);

    while (ws.hasMessage()) {
        var msg = ws.recv();
        print("Эхо: \(msg)");
    }

    ws.close();
} else {
    print("Ошибка: \(ws.lastError())");
}

Пример: Получение бинарных данных

loadPlugin("plugins/websocket.dll");

var ws = WebSocket();
ws.connect("wss://echo.websocket.events");

// Отправка бинарных данных
var data = ByteArray();
for (var i = 0; i < 256; i++) {
    data.push(i);
}
ws.sendBinary(data);

system.sleep(1000);

var received = ws.recvBinary();
if (received != null) {
    print("Получено байт: \(received.size())");
}

ws.close();

Пример: Оработка ошибок

loadPlugin("plugins/websocket.dll");

var ws = WebSocket();
ws.setTimeout(5000);

if (!ws.connect("wss://invalid-host.example.com")) {
    print("Ошибка подключения: \(ws.lastError())");
    return;
}

ws.send("test");

if (ws.recv() == null) {
    print("Нет ответа. Ошибка: \(ws.lastError())");
}

ws.close();

Пример: Библиотека без SSL (только ws://)

Для подключения к серверам без шифрования:

var ws = WebSocket();
ws.connect("ws://localhost:9001");

Для подключения к серверам с SSL:

var ws = WebSocket();
ws.connect("wss://secure-server.com");
