Загрузка плагина

loadPlugin("plugins/registry.dll");

Два способа использования:

  • Объектный — открыть ключ, читать/писать через методы, закрыть
  • Функциональный — одиночные операции одной строкой

// Прямое чтение строки
var path = RegistryReadString("HKLM", "SOFTWARE\\MyApp", "InstallPath");

// Прямая запись числа
RegistryWriteInt("HKCU", "SOFTWARE\\MyApp", "Counter", 42);

// Проверка существования
if (RegistryKeyExists("HKLM", "SOFTWARE\\MyApp")) {
    print("Приложение установлено");
}

// Перечисление подключей
var keys = RegistryEnumKeys("HKLM", "SOFTWARE\\Microsoft\\Windows\\CurrentVersion");
for (var k : keys) print(k);

Объектный интерфейс

Рекомендуется для множественных операций — открываете ключ один раз.

Создание

// Открытие на чтение
var reg = RegistryOpen("HKLM", "SOFTWARE\\ODBC\\ODBCINST.INI", "read");

// Открытие с правами записи
var reg = RegistryOpen("HKCU", "SOFTWARE\\MyApp", "all");

// Создание ключа если не существует
var reg = RegistryOpen("HKCU", "SOFTWARE\\MyApp", "all", 1);

Примеры

Чтение Installed software

var reg = RegistryOpen("HKLM", "SOFTWARE\\Microsoft\\Windows\\CurrentVersion\\Uninstall");
if (reg.keyExists()) {
    var keys = reg.enumKeys();
    for (var k : keys) {
        var sub = RegistryOpen("HKLM", "SOFTWARE\\Microsoft\\Windows\\CurrentVersion\\Uninstall\\" + k);
        var name = sub.readString("DisplayName");
        if (!name.isnull()) print(name);
        sub.close();
    }
    reg.close();
}

Сохранение настроек приложения

// Сохранение
var reg = RegistryOpen("HKCU", "SOFTWARE\\MyApp\\Settings", "all", 1);
reg.writeString("theme", "dark");
reg.writeInt("width", 1280);
reg.writeInt("height", 720);
reg.writeMultiString("recent", #[
    "file1.txt",
    "file2.txt",
    "C:\\Documents\\data.csv"
]);
reg.close();

// Загрузка
reg = RegistryOpen("HKCU", "SOFTWARE\\MyApp\\Settings");
if (reg.keyExists()) {
    var theme = reg.readString("theme");
    var w = reg.readInt("width");
    var h = reg.readInt("height");
    var files = reg.readMultiString("recent");
    reg.close();
}

Чтение PATH с раскрытием переменных

var reg = RegistryOpen("HKLM", "SYSTEM\\CurrentControlSet\\Control\\Session Manager\\Environment");
var path = reg.readExpandString("Path");
print(path);  // "C:\Windows\system32;C:\Windows;..." (переменные раскрыты)
reg.close();

Бинарные данные

var reg = RegistryOpen("HKCU", "SOFTWARE\\MyApp", "read");
var data = reg.readBinary("RawData");
print(data.size());  // размер в байтах
reg.close();

Удаление дерева ключей

var reg = RegistryOpen("HKCU", "SOFTWARE", "all");
reg.deleteTree("OldApp");  // удалит OldApp и все подключи рекурсивно
reg.close();

Определение типа значения

var reg = RegistryOpen("HKCU", "SOFTWARE\\MyApp");
var values = reg.enumValues();
for (var v : values) {
    print(v.name + " = " + v.typeName);  // "theme = string", "width = dword"...
}
reg.close();
