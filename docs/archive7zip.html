Зависимости

Требуется установленный 7z.exe (или 7z на Linux). По умолчанию плагин ищет 7z.exe в PATH.

Загрузка плагина

loadPlugin("plugins/archive7zip.dll");

Создание объекта

// С путём по умолчанию (7z.exe в PATH)
var zip = archive7zip();

// С явным путём к 7z.exe
var zip = archive7zip("C:\\Program Files\\7-Zip\\7z.exe");

list(archive, [password])

Возвращает список файлов в архиве с метаданными.

Поле кортежаТипОписание
pathstringПуть файла внутри архива
sizeintРазмер в байтах
modifiedstringДата модификации
crcstringКонтрольная сумма CRC
encryptedint (0/1)Зашифрован ли файл
var zip = archive7zip();
var files = zip.list("archive.zip");
for (var f : files) {
    print(f.path + " — " + f.size + " байт, CRC=" + f.crc);
}

// С паролем
var files = zip.list("encrypted.7z", "mypassword");

extract(archive, destPath, [password], [withPaths])

Извлекает весь архив. Возвращает tuple с результатом.

ПолеТипОписание
successint (0/1)Успешность операции
exitCodeintКод возврата 7z
outputstringВывод 7z
АргументТипПо умолчаниюОписание
archivestring—Путь к архиву
destPathstring—Куда извлекать
passwordstring—Пароль
withPathsbooltrueСохранять структуру каталогов (x vs e)
var zip = archive7zip();

// Извлечение с сохранением путей
var result = zip.extract("archive.zip", "C:\\output");
print(result.success);  // 1
print(result.output);   // вывод 7z

// Извлечение без структуры каталогов
zip.extract("archive.zip", "C:\\flat", "", false);

// С паролем
zip.extract("encrypted.7z", "C:\\out", "mypassword");

extractFile(archive, fileName, destPath, [password])

Извлекает один конкретный файл из архива.

var zip = archive7zip();
var ok = zip.extractFile("archive.zip", "docs/readme.txt", "C:\\output");
print(ok);  // 1

// С паролем
zip.extractFile("encrypted.7z", "secret.txt", "C:\\out", "pass123");

create(archive, files, [format], [password], [level])

Создаёт новый архив из списка файлов.

АргументТипПо умолчаниюОписание
archivestring—Путь к создаваемому архиву
filesArray<string>—Список файлов для архивации
formatstring"zip"Формат: "zip", "7z", "tar", "gzip"
passwordstring—Пароль (включает шифрование)
levelint5Уровень сжатия 0-9 (0=без сжатия, 9=максимум)
var zip = archive7zip();

// Простой ZIP
zip.create("backup.zip", #[
    "C:\\Docs\\file1.txt",
    "C:\\Docs\\file2.txt",
    "C:\\Photos\\image.jpg"
]);

// 7z с максимальным сжатием
zip.create("archive.7z", #[
    "data1.bin",
    "data2.bin"
], "7z", "", 9);

// Зашифрованный архив
zip.create("secret.7z", #[
    "private.txt",
    "keys.dat"
], "7z", "StrongPassword123", 9);

// GZIP (один файл — ограничение формата)
zip.create("data.tar.gz", #"data.tar", "gzip");

Ограничения

  • GZIP: формат поддерживает только один файл. Передача нескольких файлов вызовет ошибку.
  • -mhe=on (шифрование метаданных): работает только для формата 7z, не для ZIP.
  • List парольных архивов: 7z может показать список файлов парольного ZIP-архива даже без пароля (метаданные не зашифрованы). Для 7z-архивов пароль обязателен.

test(archive, [password])

Проверяет целостность архива (команда 7z t).

var zip = archive7zip();
var ok = zip.test("backup.zip");
if (ok) {
    print("Архив целый!");
} else {
    print("Архив повреждён!");
}

// С паролем
zip.test("encrypted.7z", "password");

info(archive)

Получает информацию об архиве. Возвращает tuple {type} или null.

var zip = archive7zip();
var info = zip.info("archive.zip");
print(info.type);  // "zip"

info = zip.info("backup.7z");
print(info.type);  // "7z"

Примеры

Бэкап файлов

loadPlugin("plugins/archive7zip.dll");

var zip = archive7zip();
var today = datetime.now().strftime("%Y%m%d");
var archiveName = "backup_" + today + ".7z";

var files = file.listdir("C:\\Important");
zip.create(archiveName, files, "7z", "", 9);
print("Создан: " + archiveName);

Распаковка архива

var zip = archive7zip();
var files = zip.list("download.zip");
print("Файлов в архиве: " + files.size());

var result = zip.extract("download.zip", "C:\\Downloads\\extracted");
if (result.success) {
    print("Извлечено успешно");
} else {
    print("Ошибка: " + result.output);
}

Извлечение одного файла

var zip = archive7zip();
zip.extractFile("project.7z", "src/main.cpp", "C:\\MyProject");

Проверка и инфо

var zip = archive7zip();
var info = zip.info("data.7z");
print("Формат: " + info.type);

if (zip.test("data.7z")) {
    print("Архив целый");
} else {
    print("Архив повреждён!");
}

Кастомный путь к 7z

var zip = archive7zip("D:\\Tools\\7-Zip\\7z.exe");
zip.tempPath = "D:\\Temp";
var files = zip.list("archive.zip");
