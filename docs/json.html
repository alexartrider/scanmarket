Модуль json

Работа с JSON — парсинг, сериализация и валидация (rapidjson)

RUEN

Главная / json

Доступ

Модуль json доступен глобально без uses:

print(typeName(json));  // object

Функции

ФункцияОписание
json.parse(string)Разбор JSON-строки. Объекты {} → tuple, массивы [] → array
json.load(path, [opts])Прочитать файл (кодировка определяется автоматически, BOM-safe) и разобрать. При ошибке кидает исключение с позицией. Аналог json.load(open(...)) в Python
json.dump(path, value, [opts])Сериализовать и атомарно записать в файл (по умолчанию UTF-8 без BOM). opts: {"pretty","indent","enc","overwrite","backup","verify"}. Аналог json.dump в Python + атомарность
json.stringify(value)Значение → JSON-строка (компактный вид)
json.stringify(value, true)Красивый вывод с отступами (4 пробела)
json.stringify(value, true, indent)Красивый вывод с указанным отступом (количество пробелов)
json.validate(string)Проверка валидности JSON → true / false
json.parseEx(string)Разбор с диагностикой: #{"ok","value","error"}; при ошибке error = #{"offset","line","column","message","fragment"}
json.getPath(value, path, [def])Достать значение по пути "a.b[2].c"; нет пути → def или null
json.hasPath(value, path)Существует ли путь → true / false
json.setPath(value, path, newValue)Возвращает новый глубокий клон корня: исходный объект не мутируется; промежуточные объекты создаются
json.removePath(value, path)Возвращает новый клон корня без ключа/элемента по пути; отсутствующий путь — no-op
json.validateStructure(value, schema, [label])Проверка структуры: #{"ok","errors"}. Схема — объект «ключ: тип»; суффикс ? на ключе — необязательное поле

Соответствие типов

JSONScriptПримечание
object {}tuple #{"key": val}JSON-объекты становятся кортежами
array []array #[...]JSON-массивы — обычные массивы
stringstring
number (int)int
number (float)double
true / falseint (1 / 0)Булевы значения хранятся как int

⚠ Булевы значения (true/false) при сериализации выводятся как 1/0, поскольку в скрипте нет отдельного типа bool — они хранятся как int.

Примеры

Парсинг

var j = json.parse("{\"name\": \"Ivan\", \"age\": 30}");
print(j.name);  // Ivan
print(j.age);   // 30
print(typeName(j));  // tuple

Чтение и запись файлов (load / dump)

// Прочитали и разобрали одним вызовом:
var cfg = json.load("app.json");
cfg = json.setPath(cfg, "window.title", "Название");

// Изменили и атомарно записали одним вызовом (pretty + backup):
json.dump("app.json", cfg, #{"pretty": 1, "backup": 1});

// Ошибки — исключения с позицией:
try { json.load("no.json"); } catch (e) { print(e); }   // file not found
try { json.load("bad.json"); } catch (e) { print(e); }   // parse error at 3:1: …

Парсинг с вложенными объектами

var data = json.parse("{
    \"user\": {
        \"name\": \"Petr\",
        \"scores\": [10, 20, 30]
    }
}");
print(data.user.name);       // Petr
print(data.user.scores[1]);  // 20

Сериализация

var obj = #{"name": "Ivan", "age": 30};
print(json.stringify(obj));       // {"age":30,"name":"Ivan"}
print(json.stringify(obj, true)); // красивый вывод

Сериализация с кастомным отступом

print(json.stringify(obj, true, 2));  // 2 пробела

Валидация

print(json.validate("{\"a\": 1}"));  // true
print(json.validate("{broken}"));   // false

Диагностика ошибки разбора (parseEx)

var r = json.parseEx("{\n  \"a\": 1,\n  \"b\":\n}");
print(r.ok);                        // 0
print(r.error.offset);              // 22 — позиция в символах
print(r.error.line + ":" + r.error.column);
print(r.error.fragment);            // строка-фрагмент вокруг ошибки
print(r.error.message);

Чтение по пути (getPath / hasPath)

var j = json.parse("{\"user\": {\"scores\": [10, 20]}}");
print(json.getPath(j, "user.scores[1]"));      // 20
print(json.getPath(j, "user.missing", 0));    // 0 — default
print(json.getPath(j, "user.missing"));       // null
print(json.hasPath(j, "user.scores[0]"));     // true

Изменение по пути (setPath / removePath)

var j = json.parse("{\"user\": {\"name\": \"Petr\"}}");
var j2 = json.setPath(j, "user.name", "Ivan");   // новый клон
var j3 = json.setPath(j, "user.age", 30);        // промежуточные объекты создаются
var j4 = json.removePath(j, "user.name");         // ключ удалён
print(j.user.name);   // Petr — исходник НЕ изменён
print(j2.user.name);  // Ivan

Проверка структуры (validateStructure)

var schema = #{
    "code":   "tuple",
    "form":   "tuple",
    "label?": "string"    // '?' на ключе — необязательное поле
};
var v = json.validateStructure(data, schema);
print(v.ok);                    // 0 / 1
print(json.stringify(v.errors));

Массив

var arr = json.parse("[1, 2, 3]");
print(typeName(arr));  // array
print(arr[0]);         // 1
