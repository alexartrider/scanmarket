Загрузка плагина

Два способа использования:

В скриптах функции доступны как методы объекта Crypto(). Плагин также регистрирует сервисы crypto.* для межплагинных вызовов через C++ host API — это не скриптовое пространство имён. Например, c.sha1("") возвращает SHA-1 пустой строки, а c.rsaGenerate() — tuple с ключами private и public.

// Способ 1: через объект
if (!loadPlugin("plugins/crypto.dll")) { system.exit(1); }
var c = Crypto();
var hash = c.sha256("Hello");

// Способ 2: через глобальный сервис
if (!loadPlugin("plugins/crypto.dll")) { system.exit(1); }
var hash = crypto.sha256("Hello");

Хеш-функции

МетодАргументыВозвращаетОписание
md5data, [asHex]string / bytesMD5 (128 бит)
sha1data, [asHex]string / bytesSHA-1 (160 бит)
sha256data, [asHex]string / bytesSHA-256 (256 бит)
sha512data, [asHex]string / bytesSHA-512 (512 бит)
sha3_256data, [asHex]string / bytesSHA3-256 (Keccak)

По умолчанию возвращают hex-строку. asHex=false — возвращают ByteArray.

// Хеш строки
print(crypto.md5("Hello"));      // "8b1a9953c4611296a827abf8c47804d7"
print(crypto.sha256("Hello"));    // "185f8db32271fe25f561a6fc938b2e26..."

// Хеш файла
var data = file.readbytes("image.png");
print(crypto.sha256(data));

// В виде ByteArray (не hex)
var hash = crypto.sha256("test", false);
print(hash.size());  // 32 байта

HMAC и проверка подписи

МетодАргументыВозвращаетОписание
hmackey, data, [algo], [asHex]string / bytesHMAC-подпись. algo: sha256 (по умолчанию), sha512, sha1, md5
hmacVerifykey, data, signature, [algo]int (0/1)Проверка HMAC-подписи
pbkdf2password, salt, [iterations], [keyLen], [algo]bytesДеривация ключа. iterations=100000, keyLen=32, algo=sha256
// HMAC-подпись
var key = "secret_key";
var sig = crypto.hmac(key, "important data");
print(sig);  // hex-строка

// Проверка подписи
var ok = crypto.hmacVerify(key, "important data", sig);
print(ok);  // 1

// Неверная подпись
ok = crypto.hmacVerify(key, "tampered data", sig);
print(ok);  // 0

// PBKDF2 — деривация ключа из пароля
var salt = crypto.randomBytes(16);
var key1 = crypto.pbkdf2("password", salt, 100000, 32);
var key2 = crypto.pbkdf2("password", salt, 100000, 32);
print(key1.toString() == key2.toString());  // 1 (детерминирован)

AES-шифрование (GCM)

МетодАргументыВозвращаетОписание
aesEncryptdata, key, [iv], [aad]bytesШифрование AES-GCM. Формат: IV(12) + Tag(16) + Ciphertext
aesDecryptdata, key, [aad], [asString]bytes / stringДешифрование. asString=true — вернуть строку

Ключи длиной не 16/24/32 байта автоматически хешируются через SHA-256.

// Шифрование
var key = "my_secret_key";
var encrypted = crypto.aesEncrypt("Секретное сообщение", key);
print(encrypted.size());  // IV(12) + Tag(16) + данные

// Дешифрование
var decrypted = crypto.aesDecrypt(encrypted, key, "", true);
print(decrypted);  // "Секретное сообщение"

// С AAD (Additional Authenticated Data)
var aad = "header_context";
var enc = crypto.aesEncrypt("данные", key, "", aad);
var dec = crypto.aesDecrypt(enc, key, aad, true);
print(dec);  // "данные"

// Дешифрование неверным ключом — ошибка
var bad = crypto.aesDecrypt(enc, "wrong_key", aad);
print(bad.isnull());  // true

RSA

МетодАргументыВозвращаетОписание
rsaGenerate[bits]tupleГенерация ключей. bits: 2048 (по умолч.), 4096. Возвращает {private, public} PEM
rsaEncryptdata, publicKeybytesШифрование открытым ключом (PKCS1 OAEP)
rsaDecryptdata, privateKeystringДешифрование приватным ключом
rsaSigndata, privateKey, [algo]bytesЦифровая подпись. algo: sha256 (по умолчанию)
rsaVerifydata, signature, publicKey, [algo]int (0/1)Проверка подписи
// Генерация ключей
var keys = crypto.rsaGenerate();
print(keys.private);  // PEM приватного ключа
print(keys.public);   // PEM открытого ключа

// Шифрование/дешифрование
var enc = crypto.rsaEncrypt("Secret message", keys.public);
var dec = crypto.rsaDecrypt(enc, keys.private);
print(dec);  // "Secret message"

// Цифровая подпись
var data = "Important document";
var sig = crypto.rsaSign(data, keys.private);
print(crypto.rsaVerify(data, sig, keys.public));  // 1
print(crypto.rsaVerify("tampered", sig, keys.public));  // 0

OTP (одноразовые пароли)

МетодАргументыВозвращаетОписание
totpsecret, [digits], [period]stringTime-based OTP. digits=6, period=30 сек
hotpsecret, counter, [digits]stringHMAC-based OTP. counter — счётчик

secret — строка в Base32 кодировке.

// TOTP — одноразовый пароль
var secret = "JBSWY3DPEHPK3PXP";  // Base32
var code = crypto.totp(secret);
print(code);  // "123456" (6 цифр)

// HOTP —.counter-based
var code1 = crypto.hotp(secret, 0);
var code2 = crypto.hotp(secret, 1);

JWT (JSON Web Token)

МетодАргументыВозвращаетОписание
jwtCreatepayload, key, [algo], [header]stringГенерация JWT-токена
jwtVerifytoken, key, [algo]string / nullВерификация подписи + проверка exp/nbf. Возвращает payload или null
jwtDecodetokentupleДекодирование без проверки. Возвращает {header, payload, signature}

Аргументы:

  • payload — JSON-строка или tuple (автоматически сериализуется в JSON)
  • key — секрет (строка/bytes) для HMAC, PEM-ключ приватного ключа для RSA/ECDSA/EdDSA
  • algo — алгоритм подписи (по умолчанию HS256)
  • header — tuple с дополнительными заголовочными полями (optional)

Поддерживаемые алгоритмы:

ТипАлгоритмыКлюч
HMACHS256, HS384, HS512Секретная строка или байты
RSARS256, RS384, RS512PEM-приватный ключ (создание) / PEM-открытый ключ (верификация)
ECDSAES256, ES384, ES512PEM-приватный ключ (создание) / PEM-открытый ключ (верификация)
EdDSAEdDSAPEM-приватный ключ (создание) / PEM-открытый ключ (верификация)

Валидация CLAIMS: при верификации автоматически проверяются:

  • exp (expiration time) — токен отклонён, если текущее время больше exp
  • nbf (not before) — токен отклонён, если текущее время меньше nbf
// === HS256 (секретная строка) ===
var token = crypto.jwtCreate(
     '{"sub":"user1","name":"Иван"}',
    "my_secret"
);
print(token);  // "eyJhbGciOiJI..."

// Верификация
var payload = crypto.jwtVerify(token, "my_secret");
print(payload);  // '{"sub":"user1","name":"Иван"}'

// Неверный ключ — null
print(crypto.jwtVerify(token, "wrong_key").isnull());  // true

// === Payload как tuple (автосериализация) ===
var token2 = crypto.jwtCreate(
    #{"sub": "user1", "role": "admin", "iat": 1700000000},
    "secret"
);

// === HS384 / HS512 ===
var token = crypto.jwtCreate("{}", "key", "HS512");

// === RS256 (RSA-подпись) ===
var keys = crypto.rsaGenerate();
var token = crypto.jwtCreate(
    '{"sub":"123"}',
    keys.private,
    "RS256"
);
var payload = crypto.jwtVerify(token, keys.public, "RS256");

// === ES256 (ECDSA-подпись) ===
// EC-ключи подаются готовым PEM (генерации EC в плагине нет):
// openssl ecparam -name prime256v1 -genkey -noout -out ec-private.pem
// openssl ec -in ec-private.pem -pubout -out ec-public.pem
var ecPrivateKey = file.readbytes("ec-private.pem");
var ecPublicKey  = file.readbytes("ec-public.pem");
var token = crypto.jwtCreate("{}", ecPrivateKey, "ES256");
var payload = crypto.jwtVerify(token, ecPublicKey, "ES256");

// === EdDSA (Ed25519) ===
var keys = crypto.ed25519Generate();
var token = crypto.jwtCreate('{"data":"test"}', keys.private, "EdDSA");
var payload = crypto.jwtVerify(token, keys.public, "EdDSA");

// === Дополнительные заголовки ===
var token = crypto.jwtCreate(
    "{}", "secret", "HS256",
    #{"kid": "key_001", "x5c": "cert_data"}
);

// === Автоматическая проверка exp ===
var now = datetime.timestamp(datetime.now());
var token = crypto.jwtCreate(
    #{"sub": "user1", "exp": now + 3600},
    "secret"
);
print(crypto.jwtVerify(token, "secret").isnull());  // false (ещё не истёк)

// === Декодирование без верификации ===
var parts = crypto.jwtDecode(token);
print(parts.header);     // '{"alg":"HS256","typ":"JWT"}'
print(parts.payload);    // '{"sub":"user1",...}'
print(parts.signature);  // "base64url_подпись"

Пример: шифрование файла

if (!loadPlugin("plugins/crypto.dll")) { system.exit(1); }

var key = "мой_секретный_ключ";

// Чтение файла
var data = file.readbytes("document.pdf");

// Шифрование
var encrypted = crypto.aesEncrypt(data, key);

// Сохранение зашифрованного
file.writebytes("document.pdf.enc", encrypted);
print("Зашифровано: " + encrypted.size() + " байт");

// Дешифрование
var loaded = file.readbytes("document.pdf.enc");
var decrypted = crypto.aesDecrypt(loaded, key);
file.writebytes("document_restored.pdf", decrypted);
print("Дешифровано!");

Пример: аутентификация по HMAC

var secret = "server_secret_key";

// Клиент: подпись запроса
var request = "amount=100&to=user2";
var sig = crypto.hmac(secret, request);

// Сервер: проверка
var ok = crypto.hmacVerify(secret, request, sig);
if (ok) {
    print("Запрос подлинный!");
} else {
    print("ПОДДЕЛКА!");
}

Пример: безопасное хранение пароля

var password = "user_password";

// Генерация соли
var salt = crypto.randomBytes(16);

// Деривация ключа
var hash = crypto.pbkdf2(password, salt, 100000, 32);

// Сохранение: соль + хеш
print("Соль: " + crypto.hexEncode(salt));
print("Хеш: " + crypto.hexEncode(hash));

// Проверка пароля
var check = crypto.pbkdf2(password, salt, 100000, 32);
print(check.toString() == hash.toString());  // 1
