Модуль db — Работа с базами данных

Доступ к реляционным БД через драйверы Qt (QSQLITE, QPSQL, QMYSQL, QODBC, QIBASE)

RUEN

Главная / База данных

Рекомендуемый путь к СУБД — этот модуль. Плагины mssql.dll, postgresql.dll (и нативный sqlite) — упрощённые, начального уровня; полноценно работайте здесь: SQLite — db.QSQLITE, PostgreSQL — db.QPSQL, SQL Server — db.QODBC.

Методы управления соединением

МетодОписание
db.connect(driver, connectionString, [options])Установка соединения с БД; options — строка опций Qt (например "QSQLITE_OPEN_READONLY")
db.disconnect()Закрытие соединения (откат незавершённых транзакций)
db.query()Создание объекта запроса (BuiltinDbQuery)

Методы транзакций

МетодОписание
db.begin()Начало транзакции
db.commit()Фиксация транзакции
db.rollback()Откат транзакции

Методы вставки данных

МетодОписание
db.insert(table, data, [callback, counter])Построчный INSERT (1 SQL на строку)
db.insertpacket(table, data, [callback, counter])Пакетный INSERT ... VALUES (...), (...), (...)
db.insertBatch(table, fields, rows, [callback, counter])Bulk INSERT по массиву полей + массиву значений

Примеры

// SQLite
db.connect(db.QSQLITE, "mydb.sqlite");

// PostgreSQL
db.connect(db.QPSQL, "host=localhost port=5432 dbname=mydb user=postgres");

// SQLite только для чтения (режим "read-only", write отсутствует)
db.connect(db.QSQLITE, "mydb.sqlite", "QSQLITE_OPEN_READONLY");

// Транзакция
db.begin();
db.insert("users", #{"name": "Иван", "age": 30});
db.commit();

// insertpacket — пакетная вставка
db.insertpacket("logs", #[
    #{"msg": "start", "ts": datetime.now()},
    #{"msg": "end",   "ts": datetime.now()}
]);

// insertBatch — bulk вставка
var fields = #["id", "name"];
var rows = #[ #[1, "A"], #[2, "B"], #[3, "C"] ];
db.insertBatch("t1", fields, rows);

// Прогресс-коллбэк
db.insertpacket("big_table", hugeArray, function(count) {
    print("Вставлено: \(count)");
}, 1000);

Доступ к полям как свойствам

q.open("SELECT * FROM users");
while (!q.eof()) {
    print(q.name + " — " + q.age);
    q.next();
}

Пример

var q = db.query();
q.open("SELECT name, age FROM users WHERE age > :min");
while (!q.eof()) {
    print(q.field("name") + ": " + q.field("age"));
    q.next();
}

// Через свойства
q.open("SELECT * FROM users");
var all = q.rows();  // массив tuple'ов
