SNMP модуль

Встроенный модуль для мониторинга сетевого оборудования — SNMP v1/v2c запросы, обход дерьев (walk), чтение и запись значений

RUEN

Главная / SNMP

Зависимости

Встроен в движок. Использует UDP сокеты (WSAStartup на Windows). ASN.1/BER кодирование реализовано на чистом C++ без внешних библиотек.

Создание сессии

// snmp — глобальный объект-модуль
var s = snmp.open();

Структура ответа

Методы get, getnext, getbulk, set возвращают кортеж:

#{
    "version":    1,         // 0 = SNMPv1, 1 = SNMPv2c
    "community":  "public",
    "request_id": 1,
    "error_status": 0,       // 0 = noError
    "error_index": 0,
    "pdu":        "GetResponse",
    "varbinds": [
        #{
            "oid":   "1.3.6.1.2.1.1.1.0",
            "type":  "STRING",
            "value": "RouterOS CCR1009-7G-1C-1S+"
        }
    ]
}

Примеры

Базовый GET

var s = snmp.open();
s.open("192.168.1.1", 161, "public");

// SysDescr
var r = s.get("1.3.6.1.2.1.1.1.0");
print(r.varbinds[0].value);

// SysName
r = s.get("1.3.6.1.2.1.1.5.0");
print(r.varbinds[0].value);

s.close();

GET для нескольких OID

var r = s.get(
    "1.3.6.1.2.1.1.1.0",   // sysDescr
    "1.3.6.1.2.1.1.3.0",   // sysUpTime
    "1.3.6.1.2.1.1.5.0"    // sysName
);
for (var vb : r.varbinds) {
    print(vb.oid + " = " + vb.value);
}

Walk по интерфейсам

// Walk ifTable — описание интерфейсов
var entries = s.walk("1.3.6.1.2.1.2.2.1.2");
for (var e : entries) {
    print(e.oid + " -> " + e.value);
}

GETBULK (v2c)

// non_repeaters=0, max_repetitions=10
var r = s.getbulk(0, 10, "1.3.6.1.2.1.2.2.1.2");
for (var vb : r.varbinds) {
    print(vb.oid + " = " + vb.value);
}

SET — запись значения

// Установка sysLocation
s.set("1.3.6.1.2.1.1.6.0", "Moscow", "octet");

Полный пример — мониторинг MikroTik

var s = snmp.open();
s.open("192.168.111.1", 161, "public");
s.timeout(3000);

print("Router: " + s.get("1.3.6.1.2.1.1.1.0").varbinds[0].value);
print("Name:   " + s.get("1.3.6.1.2.1.1.5.0").varbinds[0].value);

// Walk interfaces
var ifaces = s.walk("1.3.6.1.2.1.2.2.1.2");
print("Interfaces: " + ifaces.size());
for (var e : ifaces) {
    print("  " + e.value);
}

s.close();
