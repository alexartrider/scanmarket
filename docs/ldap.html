Плагин LDAP / Active Directory

Корпоративная среда: авторизация пользователей, поиск в каталоге (WinLDAP)

RUEN

Главная / Плагины / LDAP

Загрузка плагина

loadPlugin("plugins/ldap.dll");

Реализация на системной библиотеке wldap32.dll. Работает с Active Directory и любым LDAP-сервером (OpenLDAP, FreeIPA).

Авторизация в Active Directory

Функция для проверки учётных данных пользователя (корпоративная авторизация):

var ok = LDAPAuthenticate(server, port, user, password, [secure]);
АргументОписание
serverИмя или IP контроллера домена (например "dc01.corp.local")
port389 — LDAP, 636 — LDAPS (TLS)
userФормат имени пользователя, см. ниже
passwordПароль
secure1 — соединение по TLS (рекомендуется для паролей)

Допустимые форматы user:

  • UPN: "user@corp.local" — работает по обычному LDAP (389)
  • DN: "CN=Ivanov Ivan,OU=Users,DC=corp,DC=local"
  • DOMAIN\user: "CORP\user" — рекомендуется порт 636 (LDAPS)
// Обычный LDAP, пользователь в формате UPN
if (LDAPAuthenticate("dc01.corp.local", 389, "ivanov@corp.local", "password"))
    print("Доступ разрешён");
else
    print("Отказ в доступе");

// По TLS (LDAPS, порт 636)
var ok = LDAPAuthenticate("dc01.corp.local", 636, "CORP\\ivanov", "password", 1);

Объектный интерфейс

Для поиска и чтения атрибутов удобнее объект LDAP() — соединение открывается один раз.

Создание объекта

// Без аргументов — соединение открывается позже методом open()
var ld = LDAP();

// Сразу с подключением к серверу
var ld = LDAP("dc01.corp.local");              // порт по умолчанию 389
var ld = LDAP("dc01.corp.local", 636);        // LDAPS (TLS)
var ld = LDAP("dc01.corp.local", 389, 1);      // 1 — TLS-соединение

Методы объекта

МетодАргументыВозвращаетОписание

Аргументы search

АргументОписание
baseБаза поиска (DN). Пустая строка "" — корень каталога (rootDSE)
filterLDAP-фильтр, например "(&(objectCategory=person)(sAMAccountName=ivanov))"
attrsArray имён атрибутов (необязательно; по умолчанию — все)
scope"base" / "one" / "sub" или число 0/1/2 (по умолчанию "sub")

Формат результата

Каждая найденная запись — кортеж с ключом dn и именами атрибутов:

{
    "dn":      "CN=Ivanov Ivan,OU=Users,DC=corp,DC=local",
    "sAMAccountName": "ivanov",
    "userPrincipalName": "ivanov@corp.local",
    "memberOf": #[ "CN=Domain Admins,...", "CN=IT,..." ]  // многозначный атрибут — массив
}
  • Многозначные атрибуты (memberOf, proxyAddresses, ...) — Array строк
  • Бинарные атрибуты (objectSid, objectGUID, ...) — ByteArray

Примеры

Поиск пользователей по отделу

var ld = LDAP("dc01.corp.local", 389);
if (!ld.bind("svc-script@corp.local", "password")) {
    print("Ошибка авторизации: " + ld.errorText());
    system.exit(1);
}

var users = ld.search("DC=corp,DC=local",
    "(&(objectCategory=person)(objectClass=user)(department=IT))",
    #[ "sAMAccountName", "mail", "displayName", "memberOf" ]);

print("Найдено: " + users.size());
for (var u : users)
    print(u.sAMAccountName + "  " + u.mail);

Получение корня каталога (rootDSE)

var root = ld.search("", "(objectClass=*)", #[ "defaultNamingContext" ], "base");
if (root.size() > 0)
    var baseDN = root[0].defaultNamingContext;  // "DC=corp,DC=local"

Авторизация + проверка членства в группе

// Шаг 1: авторизация
if (!LDAPAuthenticate(server, 389, user, password)) {
    print("Неверный логин или пароль");
    system.exit(1);
}

// Шаг 2: проверка группы (прямые memberOf)
var ld = LDAP(server, 389);
ld.bind(user, password);
var res = ld.search("DC=corp,DC=local",
    "(&(objectClass=user)(sAMAccountName=" + ld.escapeFilter(login) + "))",
    #[ "memberOf" ]);
var isAdmin = 0;
for (var g : res[0].memberOf) {
    if (g.pos("CN=Domain Admins") >= 0) { isAdmin = 1; break; }
}
ld.close();

Экранирование ввода пользователя

Всегда экранируйте значение перед подстановкой в фильтр — символы *, (, ), \ ломают синтаксис фильтра:

var name = ld.escapeFilter(userInput);   // "ab*(c)" -> "ab\2a\28c\29"
var r = ld.search(baseDN, "(&(objectCategory=person)(sAMAccountName=" + name + "))");

Полный пример

loadPlugin("plugins/ldap.dll");

var server = "dc01.corp.local";
var login  = "ivanov@corp.local";
var pass   = "password";

// Авторизация
if (!LDAPAuthenticate(server, 389, login, pass)) {
    print("Отказ в доступе");
    system.exit(1);
}
print("Пользователь авторизован");

// Подключение и поиск
var ld = LDAP(server, 389);
if (!ld.isConnected()) { print(ld.errorText()); system.exit(1); }
ld.bind(login, pass);

var root = ld.search("", "(objectClass=*)", #[ "defaultNamingContext" ], "base");
var baseDN = root[0].defaultNamingContext;

var found = ld.search(baseDN,
    "(&(objectCategory=person)(objectClass=user)(sAMAccountName=" +
        ld.escapeFilter(login) + "))",
    #[ "dn", "displayName", "mail", "memberOf" ]);

for (var u : found) {
    print("DN: " + u.dn);
    print("Имя: " + u.displayName);
    print("Почта: " + u.mail);
}
ld.close();
