SSH модуль

Встроенный модуль для удалённого управления серверами — SSH-подключения, выполнение команд, передача файлов через SCP и SFTP

RUEN

Главная / SSH

Зависимости

Встроен в движок. Использует libssh2 1.11.1 + OpenSSL 3.

Для работы на Windows рядом с бинарником должны быть: libcrypto-3-x64.dll, libssl-3-x64.dll.

Создание сессии

// ssh — глобальный объект-модуль
var session = ssh.connect();

Формат результата sftp_readdir

Каждый элемент массива — кортеж:

#{
    "name":  "file.txt",   // string — имя файла
    "size":  12345,        // int    — размер в байтах
    "mode":  33188,        // int    — POSIX-права (octal 0644)
    "mtime": 1694000000,   // int    — время модификации (unix timestamp)
    "isdir":  0            // int    — 1 если директория
}

Пример: подключение и выполнение команды

var session = ssh.connect();

if (session.connect("192.168.1.1", 22, "root", "password")) {
    print("Подключено!");

    var result = session.exec("uname -a");
    print(result);

    var hostname = session.exec("hostname");
    print("Сервер: \(hostname)");

    session.disconnect();
} else {
    print("Ошибка: \(session.last_error())");
}

Пример: загрузка файла через SCP

var session = ssh.connect();
session.connect("192.168.1.1", 22, "user", "pass");

// Скачивание удалённого файла
var data = session.scp_recv("/var/log/syslog");
print("Размер: \(size(data))");

// Отправка данных
session.scp_send("/tmp/upload.txt", "Hello from Script!");

session.disconnect();

Пример: работа с SFTP

var s = ssh.connect();
s.connect("192.168.1.1", 22, "user", "pass");

// Инициализация SFTP
s.sftp_init();

// Просмотр директории
var files = s.sftp_readdir("/home/user");
for (var f : files) {
    print("\(f.name)  \(f.size) bytes");
}

// Запись файла
var h = s.sftp_open("/tmp/test.txt", "w");
s.sftp_write(h, "Данные из скрипта\n");
s.sftp_close(h);

// Чтение файла
var r = s.sftp_open("/tmp/test.txt", "r");
var content = s.sftp_read(r, 4096);
s.sftp_close(r);
print(content);

// Информация о файле
var st = s.sftp_stat("/tmp/test.txt");
print("Размер: \(st.size)");

// Операции с файлами
s.sftp_rename("/tmp/test.txt", "/tmp/renamed.txt");
s.sftp_unlink("/tmp/renamed.txt");

// Директории
s.sftp_mkdir("/tmp/newdir");
s.sftp_rmdir("/tmp/newdir");

s.disconnect();

Пример: массовое выполнение команд

var servers = [
    #{"host": "192.168.1.10", "user": "admin", "pass": "secret"},
    #{"host": "192.168.1.11", "user": "admin", "pass": "secret"},
    #{"host": "192.168.1.12", "user": "admin", "pass": "secret"}
];

for (var srv : servers) {
    var s = ssh.connect();
    if (s.connect(srv.host, 22, srv.user, srv.pass)) {
        print("\(srv.host):");
        print(s.exec("uptime"));
        s.disconnect();
    } else {
        print("\(srv.host): FAILED");
    }
}

Пример: мониторинг диска

var s = ssh.connect();
s.connect("server.example.com", 22, "root", "pass");

// Проверка свободного места
var df = s.exec("df -h /");
print("Дисковое пространство:");
print(df);

// Проверка процессов
var ps = s.exec("ps aux | head -20");
print("\nТоп процессов:");
print(ps);

// Проверка размера директории
var du = s.exec("du -sh /var/log");
print("\nЛоги:");
print(du);

s.disconnect();

Пример: загрузка конфигурации

var s = ssh.connect();
s.connect("db-server", 22, "dba", "password");
s.sftp_init();

// Чтение конфигурационного файла
var h = s.sftp_open("/etc/my.cnf", "r");
var config = s.sftp_read(h, 65536);
s.sftp_close(h);

// Поиск нужных строк
var lines = config.splitlines();
for (var line : lines) {
    if (line.startswith("max_connections")) {
        print("Найдено: \(line)");
    }
}

s.disconnect();
