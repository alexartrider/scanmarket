Примеры

print(math.pi);                     // 3.14159...
print(math.pow(2, 10));              // 1024
print(math.sqrt(16));                // 4
print(math.mean([1, 2, 3, 4]));    // 2.5
print(math.factorial(5));            // 120
print(math.gcd(48, 18));            // 6
print(math.isclose(0.1+0.2, 0.3));  // 1
