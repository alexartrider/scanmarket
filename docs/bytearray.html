Работа с массивами байт

RUEN

Главная / ByteArray

Создание

var ba = ByteArray();           // новый пустой экземпляр
var ba = ByteArray.fromString("Привет", ByteArray.encode_utf8);
var ba = ByteArray.fromHex("48656C6C6F");
var ba = ByteArray.fromBase64("SGVsbG8=");

Примеры

var ba = ByteArray();
ba.push(72, 101, 108, 108, 111);
print(ba.toString());           // "Hello"
print(ba.size());               // 5

var ba2 = ByteArray.fromString("Привет", ByteArray.encode_utf8);
print(ba2.size());              // 12 (UTF-8)
print(ba2.hex());               // hex-представление
print(ba2.crc32());             // контрольная сумма

var ba3 = ByteArray.fromHex("48656C6C6F");
print(ba3.toString(ByteArray.encode_utf8));  // "Hello"
print(ba3[0]);                  // 72 (код 'H')
