Зависимости

Требуются OpenSSL 3: libssl-3-x64.dll, libcrypto-3-x64.dll (размещаются рядом с бинарником).

Загрузка плагина

loadPlugin("plugins/http.dll");

Создание объекта

var http = HttpClient();

get(url)

Выполняет GET-запрос. Возвращает true если получен ответ (любой статус).

АргументТипОписание
urlstringПолный URL (http://... или https://...)
var http = HttpClient();
if (http.get("https://httpbin.org/get")) {
    print(http.status());        // 200
    print(http.statusText());    // OK
    print(http.getResponse());   // тело ответа
}

post(url, [body], [contentType])

POST-запрос с телом. По умолчанию Content-Type: application/x-www-form-urlencoded.

АргументТипПо умолчаниюОписание
urlstring—Полный URL
bodystring""Тело запроса
contentTypestring"application/x-www-form-urlencoded"Content-Type
// Form data
http.post("https://httpbin.org/post", "name=Ivan&age=30");

// JSON
http.post("https://httpbin.org/post",
    "{\"key\":\"value\"}", "application/json");

put(url, [body], [contentType])

PUT-запрос. Аналогичен post().

http.put("https://httpbin.org/put", "updated data", "text/plain");

del(url)

DELETE-запрос.

http.del("https://httpbin.org/delete");

patch(url, [body], [contentType])

PATCH-запрос. Аналогичен post().

http.patch("https://httpbin.org/patch", "partial update");

head(url)

HEAD-запрос — возвращает только заголовки без тела.

http.head("https://httpbin.org/get");
print(http.getHeader("Content-Type"));  // application/json

request(method, url, [body], [contentType])

Произвольный HTTP-метод (OPTIONS, TRACE, CONNECT и др.).

http.request("OPTIONS", "https://httpbin.org/get");

download(url, filePath)

Скачивание файла на диск. Поддерживает HTTP и HTTPS с автоматическим редиректом.

АргументТипОписание
urlstringURL файла
filePathstringПуть для сохранения
if (http.download("https://example.com/file.zip", "download/file.zip")) {
    print("Размер: " + file.size("download/file.zip"));
}

Заголовки и аутентификация

// Пользовательские заголовки
http.setHeader("X-Custom", "value");
http.setHeader("Accept", "application/json");
http.get("https://httpbin.org/headers");
http.clearHeaders();  // + сбрасывает basic/bearer auth

// Basic Auth
http.setBasicAuth("admin", "secret");
http.get("https://httpbin.org/basic-auth/admin/secret");
http.clearHeaders();

// Bearer Token
http.setBearerAuth("eyJhbGciOiJIUzI1NiJ9...");
http.get("https://api.example.com/me");
http.clearHeaders();

// Custom User-Agent
http.setUserAgent("MyApp/2.0");

Таймаут и соединение

// Таймаут (по умолчанию 30000 мсек)
http.setTimeout(5000);   // установить
print(http.setTimeout()); // прочитать текущий

// Ручное управление соединением
http.connect("api.example.com", 443, true);
print(http.isconnected());  // true
http.disconnect();

Редиректы

По умолчанию плагин автоматически следует HTTP-редиректам (301, 302, 303, 307, 308) до 10 раз.

// Отключить редиректы
http.setFollowRedirects(false);
http.get("https://httpbin.org/redirect/3");
print(http.status());  // 302

// Включить обратно
http.setFollowRedirects(true);

Ответ и заголовки

http.get("https://httpbin.org/get");

// Статус
print(http.status());       // 200
print(http.statusText());   // OK

// Тело ответа
var text = http.getResponse();     // строка
var data = http.getResponseBytes(); // ByteArray

// Заголовки ответа
print(http.getHeader("Content-Type"));
var all = http.getResponseHeaders(); // кортеж

Утилиты

// URL-кодирование/декодирование
var enc = http.urlEncode("Привет мир!");  // %D0%9F%D1%80...
var dec = http.urlDecode("%D0%9F%D1%80%D0%B8%D0%B2%D0%B5%D1%82");

// MIME-тип по расширению
print(http.getMimeType("index.html"));   // text/html

print(http.getMimeType("data.json"));    // application/json
print(http.getMimeType("app.exe"));      // application/x-msdownload

// Диагностика
print(http.lastError());  // "null" если ошибок нет

Примеры

loadPlugin("plugins/http.dll");

var http = HttpClient();

if (http.get("https://httpbin.org/get")) {
    if (http.status() == 200) {
        print(http.getResponse());
    } else {
        print("HTTP Error: " + http.status() + " " + http.statusText());

} else {
    print("Connection error: " + http.lastError());

HttpServer — встроенный HTTP-сервер

Многопоточный HTTP-сервер с callback-моделью обработки запросов. Принимает подключения в фоновом потоке, скрипт обрабатывает запросы синхронно через process().

Создание объекта

var server = HttpServer();

Методы сервера

МетодАргументыВозвращаетОписание
listenportboolНачать прослушивание порта
stop—boolОстановить сервер
islistening—boolПроверка состояния прослушивания
port—intНомер текущего порта
process[timeout_ms]tuple|nullОбработать один входящий запрос (возвращает кортеж запроса или null)
respondsocketId, responseTupleboolОтправить HTTP-ответ и закрыть соединение
lastError—stringПоследняя ошибка

listen(port)

Начинает прослушивание указанного TCP-порта. Запускает фоновый поток для приёма подключений. Возвращает true при успехе.

var server = HttpServer();
if (server.listen(8080)) {
    print("Сервер запущен на порту " + server.port());
}

process(timeout_ms)

Ожидает входящий запрос в течение timeout_ms миллисекунд (по умолчанию 50). Если запрос поступил — возвращает кортеж с данными запроса. Если запросов нет — возвращает null.

Кортеж запроса содержит:

ПолеТипОписание
methodstringHTTP-метод (GET, POST, ...)
pathstringПуть запроса (/api/data)
querystringСтрока запроса (q=test&page=1)
bodystringТело запроса
headerstupleЗаголовки запроса (ключ: значение)
portintПорт сервера
socketintИдентификатор соединения (для передачи в respond)

respond(socketId, responseTuple)

Отправляет HTTP-ответ клиенту и закрывает соединение. responseTuple — кортеж с полями:

ПолеТипПо умолчаниюОписание
statusint200HTTP-статус
statusTextstring"OK"Текст статуса
bodystring""Тело ответа
headerstuple#{} Заголовки ответа

Пример: простой сервер

loadPlugin("plugins/http.dll");

var server = HttpServer();
server.listen(8080);
print("Сервер запущен на порту 8080");

var handler = function(req) {
    print(req.method + " " + req.path);
    return #{
        "status": 200,
        "statusText": "OK",
        "headers": #{"Content-Type": "text/plain"},
        "body": "Hello, World!"
    };
};

// Основной цикл обработки
while (server.islistening()) {
    var req = server.process(100);
    if (req != null) {
        var resp = handler(req);
        server.respond(req.socket, resp);
    }
}

Пример: отдельный скрипт-сервер

// server.prg — запуск: qodeqt.exe server.prg
loadPlugin("plugins/http.dll");

var server = HttpServer();
server.listen(3000);

var handler = function(req) {
    if (req.method == "GET" && req.path == "/time") {
        return #{
            "status": 200,
            "headers": #{"Content-Type": "application/json"},
            "body": "{\"time\": \"\(datetime.now())\"}"
        };
    }
    return #{
        "status": 404,
        "statusText": "Not Found",
        "body": "Not Found"
    };
};

while (server.islistening()) {
    var req = server.process(100);
    if (req != null) {
        var resp = handler(req);
        server.respond(req.socket, resp);
    }
}
