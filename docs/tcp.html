TcpClient — TCP-клиент

Создаёт TCP-клиент для подключения к серверу. Данные принимаются фоновым потоком и буферизуются.

Создание

// Анонимный клиент
var client = TcpClient();

// Именованный клиент (доступен повторным вызовом по имени)
var client = TcpClient("myClient");

connect(host, port, [family])

Подключается к TCP-серверу. Выполняет неблокирующее подключение с тайм-аутом 5 секунд. При успешном подключении запускается фоновый поток приёма данных.

АргументТипПо умолчаниюОписание
hoststring—Адрес сервера (домен или IP)
portint—Номер порта
familyintAF_INET (2)Семейство протоколов: AF_INET=IPv4, AF_INET6=IPv6
var client = TcpClient();
client.connect("127.0.0.1", 9000);
print(client.connected);  // 1

// IPv6
client.connect("::1", 9000, 23);

send(data)

Отправляет данные серверу. Принимает строку или ByteArray.

// Отправка строки
client.send("Hello!");

// Отправка ByteArray
var bytes = ByteArray();
bytes.append(0x48);  // H
bytes.append(0x49);  // I
client.send(bytes);

sendline(text)

Отправляет строку с автоматическим добавлением \n в конце.

client.sendline("Привет серверу");
// Эквивалентно: client.send("Привет серверу\n")

recv([maxBytes], [asText])

Читает данные из внутреннего буфера (не блокирует поток). Если данных нет — возвращает null.

АргументТипПо умолчаниюОписание
maxBytesint1024Максимальное количество байт для чтения
asTextboolfalsetrue — вернуть строку, false — ByteArray
// Чтение как ByteArray
var raw = client.recv(1024);
print(typeName(raw));  // "bytearray" или "null"

// Чтение как строка
var text = client.recv(1024, true);
print(text);

recvline([timeoutMs])

Читает одну строку из буфера (до \n). Блокирует поток до появления строки или истечения тайм-аута. Символ \r перед \n автоматически удаляется.

АргументТипПо умолчаниюОписание
timeoutMsint5000Тайм-аут в миллисекундах
// Чтение с тайм-аутом 3 секунды
var line = client.recvline(3000);
if (line != null) {
    print(line);
} else {
    print("Тайм-аут!");
}

getpeername()

Возвращает адрес подключённого пира в виде кортежа. Полезно для определения IP-адреса клиента на стороне сервера.

var peer = client.getpeername();
if (peer != null) {
    print(peer.ip);    // "127.0.0.1"
    print(peer.port);  // 54321
}

TcpServer — TCP-сервер

Создаёт TCP-сервер для прослушивания входящих подключений. Клиенты автоматически принимаются в фоновом потоке.

Создание

// Анонимный сервер
var server = TcpServer();

// Именованный сервер
var server = TcpServer("myServer");

listen(port, [bindAddr], [backlog])

Начинает прослушивание входящих TCP-подключений. Если сервер уже слушает — предыдущее прослушивание и все клиенты автоматически закрываются.

АргументТипПо умолчаниюОписание
portint—Номер порта
bindAddrstring"0.0.0.0"Адрес привязки (все интерфейсы / конкретный IP)
backlogint10Очередь ожидающих подключений
var server = TcpServer();
server.listen(9000);
print(server.listening);  // 1

// Только на localhost
server.listen(9000, "127.0.0.1");

// С увеличенной очередью
server.listen(9000, "0.0.0.0", 50);

setonconnect(callback)

Устанавливает обработчик, вызываемый для каждого нового клиента. Аргумент функции — объект TcpClient подключённого клиента.

Важно: обработчик вызывается в фоновом потоке accept. Не блокируйте его надолго.

var server = TcpServer();
server.listen(9000);

server.setonconnect(function(client) {
    print("Новый клиент!");
    print("Адрес: " + client.getpeername().ip);

    // Приветствие
    client.sendline("Добро пожаловать!");

    // Чтение данных
    var line = client.recvline(5000);
    if (line != null) {
        print("Ответ: " + line);
        client.sendline("Получено: " + line);
    }
});

broadcast(data)

Отправляет данные всем подключённым клиентам одновременно. Возвращает суммарное количество отправленных байт.

// Рассылка всем при новом подключении
server.setonconnect(function(client) {
    server.broadcast("Новый пользователь!");
});

// Рассылка сообщения от клиента всем
function relay(sender, msg) {
    var clients = server.clients();
    for (var c : clients) {
        c.sendline(msg);
    }
}

clients()

Возвращает массив всех подключённых клиентов (TcpClient). Автоматически удаляет отключённых.

var clients = server.clients();
print("Подключено: " + clients.size());

// Отправка всем вручную
for (var c : clients) {
    if (c.connected) {
        c.sendline("Ping");
    }
}

Примеры

Базовый эхо-сервер

loadPlugin("tcp");

var server = TcpServer();
server.listen(9000);
print("Сервер запущен на порту 9000");

var client = TcpClient();
client.connect("127.0.0.1", 9000);

var msg = client.recvline(3000);
print("От сервера: " + msg);

client.disconnect();
server.stop();

Многоклиентный чат

loadPlugin("tcp");

var server = TcpServer();
server.listen(8080);

server.setonconnect(function(client) {
    var name = client.recvline(3000);
    server.broadcast(name + " вошёл в чат");

    while (client.connected) {
        var line = client.recvline(10000);
        if (line == null) break;
        server.broadcast(name + ": " + line);
    }

    server.broadcast(name + " вышел из чата");
});

print("Чат-сервер запущен на порту 8080");

HTTP-сервер (минимальный)

loadPlugin("tcp");

var server = TcpServer();
server.listen(8080);

server.setonconnect(function(client) {
    var request = client.recvline(2000);
    if (request != null) {
        // Пропуск заголовков
        while (true) {
            var header = client.recvline(500);
            if (header == null || header == "") break;
        }

        // Ответ
        var body = "<h1>Привет, мир!</h1>";
        var response =
            "HTTP/1.1 200 OK\r\n" +
            "Content-Type: text/html; charset=utf-8\r\n" +
            "Content-Length: " + body.size() + "\r\n" +
            "Connection: close\r\n" +
            "\r\n" + body;
        client.send(response);
    }
    client.disconnect();
});

print("HTTP-сервер: http://127.0.0.1:8080");

Именованные экземпляры

loadPlugin("tcp");

// Создание именованного сервера
var s = TcpServer("main");
s.listen(8080);

// Повторный вызов — вернёт тот же объект
var same = TcpServer("main");
print(same.listening);  // 1

// Именованный клиент
var c = TcpClient("worker");
c.connect("127.0.0.1", 8080);

// Повторный вызов — вернёт тот же объект
var same_client = TcpClient("worker");
print(same_client.connected);  // 1

Работа с ByteArray (бинарный протокол)

loadPlugin("tcp");

var server = TcpServer();
server.listen(7000);

server.setonconnect(function(client) {
    // Чтение бинарных данных
    var raw = client.recv(1024);
    if (raw != null) {
        print("Получено байт: " + raw.size());
    }
});

var client = TcpClient();
client.connect("127.0.0.1", 7000);

// Отправка бинарных данных
var packet = ByteArray();
packet.append(0x01);  // команда
packet.append(0x00);  // параметр
packet.append(0xFF);  // контрольная сумма
client.send(packet);

client.disconnect();
server.stop();

Простой TCP-клиент

loadPlugin("tcp");

var client = TcpClient();
if (!client.connect("192.168.1.100", 5000)) {
    print("Не удалось подключиться!");
    return;
}

client.sendline("PING");
var response = client.recvline(5000);
if (response == "PONG") {
    print("Сервер работает!");
}

print("Пир: " + client.getpeername().ip);
client.disconnect();

Примечания

  • Неблокирующий ввод/вывод. Все сокеты работают в неблокирующем режиме. Данные принимаются фоновым потоком и буферизуются.
  • recv() — неблокирующий. Читает из буфера, возвращает null если данных нет.
  • recvline() — блокирующий. Ждёт строку до \n с указанным тайм-аутом (по умолчанию 5 сек).
  • Повторный listen() автоматически закрывает предыдущее прослушивание и всех клиентов.
  • Клиенты сервера (созданные при подключении) автоматически удаляются при отключении.
  • Именованные экземпляры возвращаются повторно при вызове TcpClient("name") / TcpServer("name").
  • Потокобезопасность. Все методы клиента и сервера потокобезопасны.
  • Очистка. При завершении скрипта все сокеты и потоки освобождаются автоматически.
