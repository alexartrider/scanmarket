INI — Чтение

iniRead(filename)

Читает INI-файл и возвращает кортеж кортежей: #{"section": #{"key": value, ...}, ...}

  • Числа автоматически преобразуются в int или double
  • Комментарии (; и #) пропускаются
  • Пустые секции/значения допускаются
var data = iniRead("config.ini");
print(data.database.host);   // "localhost"
print(data.database.port);   // 5432 (int)
print(data.server.bind);     "0.0.0.0"

iniReadValue(filename, section, key)

Читает одно конкретное значение. Возвращает null если не найдено.

var host = iniReadValue("config.ini", "database", "host");
print(host);  // "localhost"

var missing = iniReadValue("config.ini", "noexist", "key");
print(missing);  // null

INI — Запись

iniWrite(filename, data, [encoding])

Записывает кортеж секций в INI-файл. Каждая секция — кортеж значений.

АргументТипПо умолчаниюОписание
filenamestring—Путь к файлу
datatuple—#{"section": #{"key": val, ...}, ...}
encodingstringсохранить кодировку файла"utf8", "utf8bom", "utf16", "cp1251"; для нового файла по умолчанию "cp1251"
var config = #{
    "database": #{
        "host": "localhost",
        "port": 5432,
        "name": "mydb"
    },
    "server": #{
        "bind": "0.0.0.0",
        "timeout": 30
    }
};
iniWrite("config.ini", config, "utf8");

Результат:

[database]
host=localhost
port=5432
name=mydb

[server]
bind=0.0.0.0
timeout=30

iniWriteValue(filename, section, key, value, [encoding])

Записывает/обновляет одно значение. Если файл существует — обновляет in-place. Если нет — создаёт.

АргументТипОписание
filenamestringПуть к файлу
sectionstringИмя секции
keystringИмя ключа
valueanyЗначение (string/int/double/bool)
encodingstringКодировка; если не указана — сохраняется кодировка существующего файла, для нового файла — cp1251
// Создаёт файл с нуля
iniWriteValue("settings.ini", "main", "debug", true, "utf8");
iniWriteValue("settings.ini", "main", "port", 8080, "utf8");

// Обновляет существующее значение
iniWriteValue("settings.ini", "main", "port", 9090, "utf8");

// Добавляет новую секцию в существующий файл
iniWriteValue("settings.ini", "network", "timeout", 30, "utf8");

CSV — Чтение

csvRead(filename, fields, [encoding], [separator])

Читает CSV-файл и возвращает массив кортежей. Каждая строка — кортеж с ключями из заголовка.

АргументТипПо умолчаниюОписание
filenamestring—Путь к файлу
fieldsstring/array—Строка "a;b;c" или массив #[a, b, c]
encodingstring"cp1251"Кодировка
separatorstring";"Разделитель (";", ",", "\t")

Авто-парсинг значений:

  • null → пустые строки
  • int → целые числа
  • double → вещественные числа
  • true/yes → 1, false/no → 0
  • Всё остальное — строка

Полноценный RFC-подобный разбор:

  • поля в кавычках, включая многострочные (перенос строк внутри значения);
  • разделители и кавычки внутри поля; "" — экранированная кавычка;
  • переводы строк CRLF, LF, CR; BOM в начале файла;
  • пустые строки игнорируются, пустые и хвостовые поля сохраняются;
  • пробелы вокруг поля вне кавычек отбрасываются, внутри кавычек — сохраняются;
  • ключи берутся из заголовка файла; недостающие ожидаемые поля заполняются null.
// Строка полей через разделитель
var rows = csvRead("data.csv", "name;age;city", "utf8");
print(rows[0].name);   // "Ivan"
print(rows[0].age);    // 30

// Массив полей
var rows = csvRead("data.csv", #["name", "age", "city"], "utf8");

// С запятой как разделитель
var rows = csvRead("data.csv", #["x", "y"], "utf8", ",");

// С табуляцией
var rows = csvRead("data.tsv", "a;b;c", "utf8", "\t");

CSV — Запись

csvWrite(filename, fields, data, [encoding], [separator])

Записывает массив кортежей в CSV-файл. Возвращает 1 при успехе, 0 при ошибке.

АргументТипОписание
filenamestringПуть к файлу
fieldsstring/arrayСтрока "a;b;c" или массив #["a","b","c"]
dataarrayМассив кортежей #[{...}, {...}]
encodingstring"utf8", "utf8bom", "utf16", "cp1251"; не указана — сохраняется кодировка существующего файла, для нового — cp1251
separatorstringРазделитель (по умолчанию ";")
var data = #[
    #{"name": "Ivan", "age": 30, "city": "Moscow"},
    #{"name": "Anna", "age": 25, "city": "SPb"}
];

// Точка с запятой (по умолчанию)
csvWrite("data.csv", #["name", "age", "city"], data, "utf8");

// С запятой
csvWrite("data.csv", #["name", "age", "city"], data, "utf8", ",");

// Строка полей
csvWrite("data.csv", "name;age;city", data, "utf8");

CSV — Вспомогательные

csvRowCount(filename)

Возвращает количество строк данных (без заголовка).

var n = csvRowCount("data.csv");
print("Строк: " + n);

Примеры

Конфигурация приложения

loadPlugin("plugins/inicsv.dll");

var configFile = "app.ini";

// Создать конфиг
var config = #{
    "app": #{
        "name": "MyApp",
        "version": "1.0",
        "debug": true
    },
    "database": #{
        "host": "localhost",
        "port": 5432
    }
};
iniWrite(configFile, config, "utf8");

// Прочитать
var cfg = iniRead(configFile);
print(cfg.app.name);       // "MyApp"
print(cfg.database.port);  // 5432

// Изменить одно значение
iniWriteValue(configFile, "database", "port", 3306, "utf8");

Обработка CSV-данных

loadPlugin("plugins/inicsv.dll");

// Чтение данных
var users = csvRead("users.csv", "name;email;age", "utf8");
print("Пользователей: " + csvRowCount("users.csv"));

for (var user : users) {
    print(user.name + " (" + user.email + "), возраст: " + user.age);
}

// Фильтрация и запись
var adults = [];
for (var u : users) {
    if (u.age >= 18) adults.push(u);
}
csvWrite("adults.csv", "name;email;age", adults, "utf8");

CSV с запятыми

// Чтение standard CSV
var rows = csvRead("export.csv", #["id", "name", "value"], "utf8", ",");

// Запись standard CSV
csvWrite("output.csv", #["id", "name", "value"], rows, "utf8", ",");

Табличные данные

// TSV (Tab-Separated Values)
var data = csvRead("data.tsv", "col1;col2;col3", "utf8", "\t");

// Запись с авто-определением типов
var output = #[
    #{"id": 1, "name": "first", "value": 3.14},
    #{"id": 2, "name": "second", "value": 2.71}
];
csvWrite("result.tsv", "id;name;value", output, "utf8", "\t");

Форматы файлов

INI

; комментарии через ; или #
[section_name]
key1=value1
key2=42
key3=3.14

[another_section]
host=localhost
port=5432

CSV

name;age;city
Ivan;30;Moscow
Anna;25;SPb
"He said ""hello""";25;NY

Значения с кавычками, разделителями, начальными/конечными пробелами или переносами строк автоматически оборачиваются в двойные кавычки. Двойные кавычки внутри значений удваиваются: " → "". Строки разделяются CRLF.

Кодировки

ПараметрОписание
"utf8"UTF-8 без BOM
"utf8bom"UTF-8 с BOM
"utf16"UTF-16 LE
"cp1251"Windows-1251 (по умолчанию)

Чтение автоматически определяет кодировку (BOM + эвристика).

Если при записи кодировка не указана (или передана "auto") и файл уже существует — сохраняется его кодировка; для нового файла используется "cp1251". Это позволяет обновлять UTF-8/UTF-16 файлы без указания кодировки при каждом вызове.

Примечания

  • Числа: десятичный разделитель всегда точка (.), независимо от системной локали
  • Типы: пустые строки → null, целые числа → int, вещественные → double, true/yes → 1, false/no → 0
  • CSV-кавычки: значения с разделителями, кавычками или переносами строк автоматически оборачиваются в "
  • INI-комментарии: строки, начинающиеся с ; или #, пропускаются; заголовок секции может заканчиваться комментарием: [section] ; note. Внутри значений комментарии не вырезаются
