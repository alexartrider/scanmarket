Навигация по записям

МетодАргументыВозвращаетОписание
next—int (0/1)Переход к следующей записи. Возвращает 0 при достижении EOF
prev—int (0/1)Переход к предыдущей записи. Возвращает 0 при достижении BOF
rewind——Перемотка в начало (указатель перед первой записью)
moveTonint (1/0)Переход к записи по номеру (1-based)
eof—int (0/1)Достигнут ли конец файла
bof—int (0/1)Находимся ли перед первой записью
recno—intТекущий номер записи (1-based, 0 = перед первой)
recordCount—intОбщее количество записей
var db = dbffile();
db.open("C:/data/employees.dbf");

// Последовательный обход
db.rewind();
while (db.next()) {
    var row = db.currentRow();
    print(row["NAME"]);
}

// Переход к конкретной записи
db.moveTo(5);  // 5-я запись
var row = db.currentRow();
print(row);

// Позиция в файле
print("Запись: " + db.recno() + " из " + db.recordCount());
print("EOF: " + db.eof());
print("BOF: " + db.bof());

Удаление записей

МетодАргументыВозвращаетОписание
deleteRownint (1/0)Пометка записи как удалённой (1-based)
pack——Физическое удаление всех помеченных записей
isDeletednint (0/1)Проверка, помечена ли запись как удалённая
undeletenint (1/0)Восстановление удалённой записи

Важно: в DBF удаление работает в два этапа. Сначала запись помечается как удалённая (deleteRow), затем при вызове pack() помеченные записи физически удаляются.

var db = dbffile();
db.open("C:/data/employees.dbf");

// Пометить запись на удаление
db.deleteRow(5);

// Проверить статус
print(db.isDeleted(5));  // 1
print(db.isDeleted(1));  // 0

// Восстановить если нужно
db.undelete(5);
print(db.isDeleted(5));  // 0

// Пометить снова
db.deleteRow(5);

// Физически удалить все помеченные
db.pack();
print(db.recordCount());  // стало меньше

db.save();

for (var i = 0; i < names.size(); i++) {
    print("  " + names[i]);
}

// Полные описания полей
var meta = db.fields();
for (var i = 0; i < meta.size(); i++) {
    print("  " + meta[i].name

        + "(" + meta[i].length + "," + meta[i].decimals + ")");
}

// Кодировка
print("Кодировка: " + db.getEncoding());  // 1251
db.setEncoding(866);

Поиск записей

МетодАргументыВозвращаетОписание
locatefieldName, valueintПоиск по значению поля. Возвращает номер записи (1-based) или 0 если не найдено. Перемещает указатель
var db = dbffile();
db.open("C:/data/employees.dbf");

// Найти запись с именем "Иван"
var rec = db.locate("NAME", "Иван");
if (rec > 0) {
    print("Найдена запись #" + rec);
    var row = db.currentRow();
    print(row);
} else {
    print("Не найдено");
}

// Поиск с навигацией — найти все записи
db.rewind();
while (db.next()) {
    var row = db.currentRow();
    if (row["AGE"] > 30) {
        print(row["NAME"] + " — " + row["AGE"]);
    }
}

Пример: создание базы данных

if (!loadPlugin("plugins/dbf.dll")) { system.exit(1); }

var db = dbffile();

// Определение структуры
var fields = #[
    #{"name": "CODE",    "type": "N", "length": 6,  "decimals": 0},
    #{"name": "NAME",    "type": "C", "length": 50, "decimals": 0},
    #{"name": "PRICE",   "type": "N", "length": 12, "decimals": 2},
    #{"name": "INSTOCK", "type": "N", "length": 8,  "decimals": 0},
    #{"name": "ACTIVE",  "type": "L", "length": 1,  "decimals": 0}
];

db.create("C:/data/products.dbf", fields);
print("Файл создан!");

// Добавление товаров
db.append(1001, "Ноутбук ASUS",     54990.00, 15, 1);
db.append(1002, "Монитор Samsung",  23990.50, 8,  1);
db.append(1003, "Клавиатура Logi", 2490.00,  42, 1);
db.append(1004, "Мышь A4Tech",     590.00,   100,0);

db.save();
print("Данные сохранены!");
print("Записей: " + db.recordCount());  // 4

db.close();

Пример: чтение и обработка

var db = dbffile();
db.open("C:/data/products.dbf");

// Прочитать все записи
var products = db.readAll();
for (var i = 0; i < products.size(); i++) {
    var p = products[i];
    print(p["CODE"] + " | "
        + p["NAME"] + " | "
        + p["PRICE"] + " руб.");
}

// Найти товары дороже 10000
print("\nДорогие товары:");
for (var i = 0; i < products.size(); i++) {
    if (products[i]["PRICE"] > 10000) {
        print("  " + products[i]["NAME"]);
    }
}

// Структура таблицы
var meta = db.fields();
for (var i = 0; i < meta.size(); i++) {
    print(meta[i].name + ": тип=" + meta[i].type
        + " длина=" + meta[i].length);
}

db.close();

Пример: копирование данных

var db = dbffile();

// Открыть исходный файл
db.open("C:/data/source.dbf");
var data = db.readAll();
var meta = db.fields();
db.close();

// Создать новый файл с той же структурой

db.writeAll(data);
db.save();
db.close();

print("Скопировано записей: " + data.size());
