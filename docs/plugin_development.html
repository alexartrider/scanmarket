Разработка плагинов

Полное руководство по созданию плагинов для Script

RUEN

Главная / Разработка плагинов

Обзор

Плагины — это загружаемые модули (DLL/SO), расширяющие функциональность скриптового движка. Каждый плагин экспортирует набор функций, доступных из скрипта.

Два уровня API

УровеньФайлыДля кого
C APIscript_plugin_api.hЛюбой язык (C, C++, Rust, и др.) — низкоуровневый интерфейс
C++ Helperscript_plugin_helper.h + .cppC++23 — RAII-обёртки, фабрики, итераторы

Жизненный цикл плагина

  1. QodeScriptGetPlugin() — хост загружает DLL, вызывает эту функцию для получения интерфейса
  2. Initialize(host_api) — передаёт таблицу функций хоста, плагин сохраняет g_api
  3. RegisterFunctions(ctx) — плагин регистрирует свои функции и фабрики объектов
  4. Работа... — скрипт вызывает функции плагина
  5. Shutdown() — плагин выгружается, освобождает ресурсы

Быстрый старт — минимальный плагин

1. Файл проекта (plugin.pro)

CONFIG -= qt
TEMPLATE = lib
DEFINES += QSCRIPT_PLUGIN_BUILD
CONFIG += c++23

SCRIPT_ROOT = $$PWD/../../..
INCLUDEPATH += $$SCRIPT_ROOT/plugins

SOURCES += ../../../plugins/script_plugin_helper.cpp \
           plugin.cpp

           ../../../plugins/script_plugin_helper.h

CONFIG(debug, debug|release) {
    DESTDIR = $$QT_ROOT/bind/plugins
} else {
    DESTDIR = $$QT_ROOT/bin/plugins
}

2. Файл плагина (plugin.cpp)

#define QSCRIPT_PLUGIN_BUILD
#include "script_plugin_helper.h"

const QScriptHostAPI* g_api = nullptr;

// Простая функция
QScriptValueHandle SayHello(QScriptContextHandle ctx,
                            const QScriptArg* args, int arg_count)
{
    CHECK_MIN_PARAMS(1, "SayHello");
    ScriptString name = getString(args[0].value);
    ScriptString msg = Q("Привет, ") + name + Q("!");
    return g_api->CreateString(ctx, msg.c_str());
}

// Инициализация
QSCRIPT_EXPORT int QSCRIPT_API
QodeScriptInitialize(const QScriptHostAPI* host_api)
{
    if (!host_api || host_api->version < 1) return 0;
    g_api = host_api;
    return 1;
}

// Регистрация функций
QSCRIPT_EXPORT void QSCRIPT_API
QodeScriptRegisterFunctions(QScriptContextHandle ctx)
{
    g_api->RegisterFunction(ctx, Q("SayHello"), SayHello);
}

// Завершение
QSCRIPT_EXPORT void QSCRIPT_API QodeScriptShutdown(void)
{
    g_api = nullptr;
}

// Экспорт интерфейса
extern "C" {
QSCRIPT_EXPORT const QScriptPluginInterface* QSCRIPT_API
QodeScriptGetPlugin(void)
{
    static QScriptPluginInterface iface = {
        1, "MyPlugin", "Author", 1,
        QodeScriptInitialize,
        QodeScriptRegisterFunctions,
        QodeScriptShutdown
    };
    return &iface;
}

#ifdef _WIN32
int _libmain(unsigned long) { return 1; }
#endif
}

3. Использование в скрипте

loadPlugin("myplugin");
print(SayHello("Мир"));

Полный пример — объект плагина

#define QSCRIPT_PLUGIN_BUILD
#include "script_plugin_helper.h"

const QScriptHostAPI* g_api = nullptr;

class Counter : public PluginObject
{
public:
    Counter() : count_(0)
    {
        // Свойство count (только чтение)
        BIND_PROPERTY(this, "count",
            [this](QScriptContextHandle ctx) {
                return g_api->CreateInt(ctx, count_);
            }
        );

        // Метод increment
        BIND_METHOD(this, "increment", &Counter::Increment);

        // Метод reset
        BIND_METHOD(this, "reset", &Counter::Reset);
    }

    QScriptValueHandle Increment(QScriptContextHandle ctx,
                                  const QScriptArg* args, int arg_count)
    {
        CHECK_MIN_PARAMS(0, "increment");
        int64_t step = 1;
        if (arg_count >= 1)
            step = getLong(args[0].value);
        count_ += step;
        return g_api->CreateInt(ctx, count_);
    }

    QScriptValueHandle Reset(QScriptContextHandle ctx,
                              const QScriptArg* args, int arg_count)
    {
        CHECK_MIN_PARAMS(0, "reset");
        count_ = 0;
        return g_api->CreateNull(ctx);
    }

private:
    int64_t count_;
};

// Фабрика объектов
static QScriptValueHandle CreateCounter(QScriptContextHandle ctx,
                                         const QScriptArg* args, int arg_count)
{
    return CreatePluginObject(ctx, new Counter());
}

// Экспорт плагина
QSCRIPT_EXPORT int QSCRIPT_API
QodeScriptInitialize(const QScriptHostAPI* host_api)
{
    if (!host_api || host_api->version < 1) return 0;
    g_api = host_api;
    return 1;
}

QSCRIPT_EXPORT void QSCRIPT_API
QodeScriptRegisterFunctions(QScriptContextHandle ctx)
{
    g_api->RegisterFunction(ctx, Q("Counter"), CreateCounter);
}

QSCRIPT_EXPORT void QSCRIPT_API QodeScriptShutdown(void)
{
    g_api = nullptr;
}

extern "C" {
QSCRIPT_EXPORT const QScriptPluginInterface* QSCRIPT_API
QodeScriptGetPlugin(void)
{
    static QScriptPluginInterface iface = {
        1, "Counter Plugin", "Author", 1,
        QodeScriptInitialize,
        QodeScriptRegisterFunctions,
        QodeScriptShutdown
    };
    return &iface;
}
#ifdef _WIN32
int _libmain(unsigned long) { return 1; }
#endif
}

Итерация по коллекциям

Прямая итерация (low-level)

void ProcessTuple(QScriptContextHandle ctx, QScriptValueHandle tuple)
{
    QScriptValueHandle key, value;
    void* iterator = nullptr;
    QScriptResult res;

    if (g_api->TupleFirst(ctx, tuple, &key, &value, &iterator, &res))
    {
        do {
            // Обработка key, value
            ScriptValueEx k(g_api, ctx, key);
            ScriptValueEx v(g_api, ctx, value);
            Logger::Debug("Key: " + Utf16ToUtf8(k.toString(ctx)));

        } while (g_api->TupleNext(ctx, tuple, iterator, &key, &value, &res));
    }
    // iterator автоматически уничтожается при достижении конца
    // или при вызове TupleIteratorDestroy(iter) досрочно
}

Range-based for (C++ helper)

void ProcessTuple(QScriptContextHandle ctx, QScriptValueHandle tuple)
{
    for (auto& [key, value] : TupleRange(g_api, ctx, tuple))
    {
        ScriptValueEx k(g_api, ctx, key);
        ScriptValueEx v(g_api, ctx, value);
        // обработка...
    }
}

Сохранение значений за пределы итерации

ScriptValue persistent_key;
for (auto& [k, v] : TupleRange(g_api, ctx, tuple))
{
    persistent_key = ScriptValue(g_api, ctx, k);
    // persistent_key валиден после цикла
}

Межплагинное взаимодействие (сервисы)

Плагины могут предоставлять методы для вызова другими плагинами.

Регистрация сервиса

static QScriptValueHandle MyServiceFunc(
    QScriptContextHandle ctx, const QScriptArg* args,
    int arg_count, void* user_data)
{
    // Реализация сервиса
    return g_api->CreateInt(ctx, 42);
}

g_api->RegisterService(ctx,
    Q("MyService"),     // имя сервиса
    Q("Process"),       // имя метода
    MyServiceFunc,      // функция
    1,                  // минимум аргументов
    -1,                 // максимум (-1 = без ограничений)
    nullptr,            // user_data
    g_pluginId          // handle плагина
);

Вызов сервиса из другого плагина

QScriptResult res;
QScriptValueHandle result = g_api->CallService(ctx,
    Q("MyService"), Q("Process"), args, arg_count, &res);

Типичные ошибки и как их избежать

1. Хранение QScriptValueHandle

// НЕПРАВИЛЬНО:
QScriptValueHandle saved_handle;
void Func(QScriptContextHandle ctx, QScriptValueHandle h) {
    saved_handle = h;  // h будет невалиден после возврата!
}

// ПРАВИЛЬНО:
ScriptValue saved_ref;
void Func(QScriptContextHandle ctx, QScriptValueHandle h) {
    saved_ref = ScriptValue(g_api, ctx, h);  // persistent ref
}

2. Утечка памяти при итераторах

// НЕПРАВИЛЬНО — утечка если break до конца:
while (...) {
    if (cond) break;  // iterator не уничтожен!
}

// ПРАВИЛЬНО — используйте ScriptStlIterator или вызовите Destroy:
void* iter;
g_api->TupleFirst(ctx, h, &key, &val, &iter, &res);
// ... при break:
g_api->TupleIteratorDestroy(iter);

3. Неверный контекст для handle(ctx)

// НЕПРАВИЛЬНО:
QScriptValueHandle h = some_ref_to_handle;
// ... позже, в другом контексте:
ScriptValueEx val(g_api, other_ctx, h);  // h может быть невалиден!

// ПРАВИЛЬНО — всегда используйте тот же ctx:
ScriptValueEx val(g_api, ctx, ref.handle(ctx));

4. ArenaLevelScope — критическая ошибка

// НЕПРАВИЛЬНО — значение уничтожается при выходе из scope:
QScriptValueHandle HostGet(QScriptContextHandle ctx, QScriptValueHandle obj)
{
    ArenaLevelScope scope{ctx->arena};
    auto val = MakeTempValue(ctx, ScriptValue(int64_t(42)));
    return val;  // val указывает на уничтоженную память!
}

// ПРАВИЛЬНО — копируем до завершения scope:
QScriptValueHandle HostGet(QScriptContextHandle ctx, QScriptValueHandle obj)
{
    ArenaLevelScope scope{ctx->arena};
    auto val = MakeTempValue(ctx, ScriptValue(int64_t(42)));
    return val->Clone();
}

См. подробности в разделе Безопасность арены.

5. Забытая проверка аргументов

// НЕПРАВИЛЬНО:
QScriptValueHandle MyFunc(QScriptContextHandle ctx,
                           const QScriptArg* args, int arg_count)
{
    ScriptString s = getString(args[0].value);  // краш если arg_count == 0
}

// ПРАВИЛЬНО:
QScriptValueHandle MyFunc(QScriptContextHandle ctx,
                           const QScriptArg* args, int arg_count)
{
    CHECK_MIN_PARAMS(1, "MyFunc");
    ScriptString s = getString(args[0].value);
}
