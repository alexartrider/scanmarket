SNMP module

Built-in module for monitoring network devices — SNMP v1/v2c requests, tree walk, reading and writing values

RUEN

Home / SNMP

Dependencies

Built into the engine. Uses UDP sockets (WSAStartup on Windows). ASN.1/BER encoding is implemented in pure C++ with no external libraries.

Creating a session

// snmp — global module object
var s = snmp.open();

MethodArgumentsReturnsDescription

Response structure

The get, getnext, getbulk and set methods return a tuple:

#{
    "version":    1,         // 0 = SNMPv1, 1 = SNMPv2c
    "community":  "public",
    "request_id": 1,
    "error_status": 0,       // 0 = noError
    "error_index": 0,
    "pdu":        "GetResponse",
    "varbinds": [
        #{
            "oid":   "1.3.6.1.2.1.1.1.0",
            "type":  "STRING",
            "value": "RouterOS CCR1009-7G-1C-1S+"
        }
    ]
}

Examples

Basic GET

var s = snmp.open();
s.open("192.168.1.1", 161, "public");

// SysDescr
var r = s.get("1.3.6.1.2.1.1.1.0");
print(r.varbinds[0].value);

// SysName
r = s.get("1.3.6.1.2.1.1.5.0");
print(r.varbinds[0].value);

s.close();

GET for multiple OIDs

var r = s.get(
    "1.3.6.1.2.1.1.1.0",   // sysDescr
    "1.3.6.1.2.1.1.3.0",   // sysUpTime
    "1.3.6.1.2.1.1.5.0"    // sysName
);
for (var vb : r.varbinds) {
    print(vb.oid + " = " + vb.value);
}

Walk over interfaces

// Walk ifTable — interface descriptions
var entries = s.walk("1.3.6.1.2.1.2.2.1.2");
for (var e : entries) {
    print(e.oid + " -> " + e.value);
}

GETBULK (v2c)

// non_repeaters=0, max_repetitions=10
var r = s.getbulk(0, 10, "1.3.6.1.2.1.2.2.1.2");
for (var vb : r.varbinds) {
    print(vb.oid + " = " + vb.value);
}

SET — writing a value

// Setting sysLocation
s.set("1.3.6.1.2.1.1.6.0", "Moscow", "octet");

Full example — MikroTik monitoring

var s = snmp.open();
s.open("192.168.111.1", 161, "public");
s.timeout(3000);

print("Router: " + s.get("1.3.6.1.2.1.1.1.0").varbinds[0].value);
print("Name:   " + s.get("1.3.6.1.2.1.1.5.0").varbinds[0].value);

// Walk interfaces
var ifaces = s.walk("1.3.6.1.2.1.2.2.1.2");
print("Interfaces: " + ifaces.size());
for (var e : ifaces) {
    print("  " + e.value);
}

s.close();
