File operations

MethodArgumentsReturnsDescription
openpath, [encoding]int (1/0)Opens a DBF file. encoding: 1251 (default) or 866
close——Closes the file and clears the data
createpath, fields, [encoding]int (1/0)Creates a new DBF file
save[path]int (1/0)Saves to a file. path is optional (defaults to the current file)
isOpen—int (0/1)Whether the file is open
isModified—int (0/1)Whether there have been changes since the last open/save

Field format for create()

An array of tuples describing each field:

KeyTypeDescription
namestringField name (up to 10 characters)
typestringType: C (string), N (number), D (date), L (logical)
lengthintField length in bytes
decimalsintNumber of decimal places (for N)
// Opening an existing file
var db = dbffile();
var ok = db.open("C:/data/employees.dbf");
print("Opened: " + ok);

// Opening with CP866 encoding
db.open("C:/data/old_file.dbf", 866);

// Creating a new file
var fields = #[
    #{"name": "NAME",    "type": "C", "length": 30, "decimals": 0},
    #{"name": "AGE",     "type": "N", "length": 5,  "decimals": 0},
    #{"name": "SALARY",  "type": "N", "length": 10, "decimals": 2},
    #{"name": "BDATE",   "type": "D", "length": 8,  "decimals": 0},
    #{"name": "ACTIVE",  "type": "L", "length": 1,  "decimals": 0}
];
db.create("C:/data/new.dbf", fields);

// Saving
db.save();
// Or save to another file:
db.save("C:/data/backup.dbf");

Record navigation

MethodArgumentsReturnsDescription
next—int (0/1)Moves to the next record. Returns 0 when EOF is reached
prev—int (0/1)Moves to the previous record. Returns 0 when BOF is reached
rewind——Rewinds to the beginning (pointer before the first record)
moveTonint (1/0)Moves to a record by number (1-based)
eof—int (0/1)Whether the end of file has been reached
bof—int (0/1)Whether the pointer is before the first record
recno—intCurrent record number (1-based, 0 = before the first)
recordCount—intTotal number of records
var db = dbffile();
db.open("C:/data/employees.dbf");

// Sequential iteration
db.rewind();
while (db.next()) {
    var row = db.currentRow();
    print(row["NAME"]);
}

// Jump to a specific record
db.moveTo(5);  // 5th record
var row = db.currentRow();
print(row);

// Position in the file
print("Record: " + db.recno() + " of " + db.recordCount());
print("EOF: " + db.eof());
print("BOF: " + db.bof());

Reading data

row(n)tuple{"NAME":"Ivan","AGE":30}
readFlat()array["Ivan", 30, "Peter", 25, ...]

Value types in a tuple

DBF typeScript typeExample
C (Character)string"Ivan"
N (Numeric, decimals=0)int30
N (Numeric, decimals>0)double50000.00
D (Date)datetimedatetime
L (Logical)int (0/1)1 or 0
var db = dbffile();
db.open("C:/data/employees.dbf");

// Read all records
var rows = db.readAll();
print("Records: " + rows.size());
for (var i = 0; i < rows.size(); i++) {
    print(rows[i]["NAME"] + " — " + rows[i]["AGE"]);
}

// Read a single record
var row = db.row(3);  // 3rd record
print(row["NAME"]);
print(row["SALARY"]);

// Current record (after next/moveTo)
db.moveTo(1);
var cur = db.currentRow();
print(cur);

// Flat array
var flat = db.readFlat();
print("Elements: " + flat.size());  // recordCount * fieldCount

Writing data

MethodArgumentsReturnsDescription
append(val1, val2, ...) or (array) or (tuple)int (1/0)Appends a record (3 call forms)
updateRown, dataint (1/0)Updates a record by number (1-based)
updateCurrentRowdataint (1/0)Updates the current record
writeAlldataint (1/0)Replaces all records entirely

Call forms of append()

FormExample
Separate argumentsdb.append("Ivan", 30, 50000.00, dt, 1)
Arraydb.append(#["Ivan", 30, 50000.00, dt, 1])
Tupledb.append(#{"NAME":"Ivan", "AGE":30, "SALARY":50000.00})
var db = dbffile();
db.open("C:/data/employees.dbf");

// Form 1: separate arguments
var dt = datetime.makedatetime(2024, 1, 15);
db.append("Ivanov Ivan", 30, 50000.00, dt, 1);

// Form 2: array
db.append(#["Petrov Pyotr", 25, 45000.50, dt, 1]);

// Form 3: tuple (named fields)
db.append(#{
    "NAME":   "Sidorov Sidor",
    "AGE":    40,
    "SALARY": 60000.00,
    "BDATE":  datetime.makedatetime(1984, 5, 20),
    "ACTIVE": 1
});

// Update a record by number
db.updateRow(1, #{"NAME": "Ivanov I.I.", "AGE": 31});

// Update the current record
db.moveTo(2);
db.updateCurrentRow(#{"SALARY": 55000.00});

// Complete replacement of the data
var newData = #[
    #{"NAME": "New", "AGE": 20, "SALARY": 30000.00, "BDATE": dt, "ACTIVE": 1}
];
db.writeAll(newData);

// Saving
db.save();

Deleting records

MethodArgumentsReturnsDescription
deleteRownint (1/0)Marks a record as deleted (1-based)
pack——Physically removes all marked records
isDeletednint (0/1)Checks whether a record is marked as deleted
undeletenint (1/0)Restores a deleted record

Important: in DBF, deletion works in two stages. First the record is marked as deleted (deleteRow), then when pack() is called the marked records are physically removed.

var db = dbffile();
db.open("C:/data/employees.dbf");

// Mark a record for deletion
db.deleteRow(5);

// Check the status
print(db.isDeleted(5));  // 1
print(db.isDeleted(1));  // 0

// Restore if needed
db.undelete(5);
print(db.isDeleted(5));  // 0

// Mark again
db.deleteRow(5);

// Physically remove all marked records
db.pack();
print(db.recordCount());  // decreased

db.save();

Field metadata

MethodArgumentsReturnsDescription
fieldCount—intNumber of fields
fields—array of tupleField descriptions: {name, type, length, decimals}
fieldMeta—array of tupleAlias of fields()
fieldNames—array of stringArray of field names
getEncoding—intCurrent encoding (1251 or 866)
setEncodingenc—Sets the encoding (1251 or 866)
var db = dbffile();
db.open("C:/data/employees.dbf");

// Number of fields and records
print("Fields: " + db.fieldCount());
print("Records: " + db.recordCount());

// Field names
var names = db.fieldNames();
for (var i = 0; i < names.size(); i++) {
    print("  " + names[i]);
}

// Full field descriptions
var meta = db.fields();
for (var i = 0; i < meta.size(); i++) {
    print("  " + meta[i].name
        + " : " + meta[i].type
        + "(" + meta[i].length + "," + meta[i].decimals + ")");
}

// Encoding
print("Encoding: " + db.getEncoding());  // 1251
db.setEncoding(866);

Finding records

MethodArgumentsReturnsDescription
locatefieldName, valueintSearches by field value. Returns the record number (1-based) or 0 if not found. Moves the pointer
var db = dbffile();
db.open("C:/data/employees.dbf");

// Find a record with the name "Ivan"
var rec = db.locate("NAME", "Ivan");
if (rec > 0) {
    print("Found record #" + rec);
    var row = db.currentRow();
    print(row);
} else {
    print("Not found");
}

// Search with navigation — find all records
db.rewind();
while (db.next()) {
    var row = db.currentRow();
    if (row["AGE"] > 30) {
        print(row["NAME"] + " — " + row["AGE"]);
    }
}

Example: creating a database

if (!loadPlugin("plugins/dbf.dll")) { system.exit(1); }

var db = dbffile();

// Defining the structure
var fields = #[
    #{"name": "CODE",    "type": "N", "length": 6,  "decimals": 0},
    #{"name": "NAME",    "type": "C", "length": 50, "decimals": 0},
    #{"name": "PRICE",   "type": "N", "length": 12, "decimals": 2},
    #{"name": "INSTOCK", "type": "N", "length": 8,  "decimals": 0},
    #{"name": "ACTIVE",  "type": "L", "length": 1,  "decimals": 0}
];

db.create("C:/data/products.dbf", fields);
print("File created!");

// Adding products
db.append(1001, "ASUS Laptop",     54990.00, 15, 1);
db.append(1002, "Samsung Monitor",  23990.50, 8,  1);
db.append(1003, "Logi Keyboard", 2490.00,  42, 1);
db.append(1004, "A4Tech Mouse",     590.00,   100,0);

db.save();
print("Data saved!");
print("Records: " + db.recordCount());  // 4

db.close();

Example: reading and processing

var db = dbffile();
db.open("C:/data/products.dbf");

// Read all records
var products = db.readAll();
for (var i = 0; i < products.size(); i++) {
    var p = products[i];
    print(p["CODE"] + " | "
        + p["NAME"] + " | "
        + p["PRICE"] + " RUB");
}

// Find products more expensive than 10000
print("\nExpensive products:");
for (var i = 0; i < products.size(); i++) {
    if (products[i]["PRICE"] > 10000) {
        print("  " + products[i]["NAME"]);
    }
}

// Table structure
var meta = db.fields();
for (var i = 0; i < meta.size(); i++) {
    print(meta[i].name + ": type=" + meta[i].type
        + " length=" + meta[i].length);
}

db.close();

Example: deletion and packing

var db = dbffile();
db.open("C:/data/products.dbf");

// Delete the product with code 1004
var rec = db.locate("CODE", "1004");
if (rec > 0) {
    db.deleteRow(rec);
    print("Record #" + rec + " marked for deletion");
}

// Check
print("Deleted: " + db.isDeleted(rec));  // 1

// Restore if you made a mistake
// db.undelete(rec);

// Physically remove
db.pack();
print("Records remaining: " + db.recordCount());

db.save();
db.close();

Example: copying data

var db = dbffile();

// Open the source file
db.open("C:/data/source.dbf");
var data = db.readAll();
var meta = db.fields();
db.close();

// Create a new file with the same structure
db.create("C:/data/copy.dbf", meta);
db.writeAll(data);
db.save();
db.close();

print("Records copied: " + data.size());

Error handling

var db = dbffile();

// Check opening
if (!db.open("C:/data/missing.dbf")) {
    print("File not found or corrupted!");
    system.exit(1);
}

// Check after operations
var ok = db.append("test", 123);
if (!ok) {
    print("Write error!");
}

// Check reading
var row = db.row(999);
if (row.isnull()) {
    print("Record not found!");
}

// Check the current state
print("Open: " + db.isOpen());
print("Modified: " + db.isModified());

db.close();
