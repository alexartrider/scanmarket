Overview

Plugins are loadable modules (DLL/SO) that extend the functionality of the script engine. Each plugin exports a set of functions available from scripts.

Two API levels

LevelFilesWho it is for
C APIscript_plugin_api.hAny language (C, C++, Rust, etc.) — low-level interface
C++ Helperscript_plugin_helper.h + .cppC++23 — RAII wrappers, factories, iterators

Plugin lifecycle

  1. QodeScriptGetPlugin() — the host loads the DLL and calls this function to obtain the interface
  2. Initialize(host_api) — passes the host function table; the plugin stores g_api
  3. RegisterFunctions(ctx) — the plugin registers its functions and object factories
  4. Running... — the script calls the plugin functions
  5. Shutdown() — the plugin is unloaded and releases its resources

Quick start — minimal plugin

1. Project file (plugin.pro)

CONFIG -= qt
TEMPLATE = lib
DEFINES += QSCRIPT_PLUGIN_BUILD
CONFIG += c++23

SCRIPT_ROOT = $$PWD/../../..
INCLUDEPATH += $$SCRIPT_ROOT/plugins

SOURCES += ../../../plugins/script_plugin_helper.cpp \
           plugin.cpp
HEADERS += ../../../plugins/script_plugin_api.h \
           ../../../plugins/script_plugin_helper.h

CONFIG(debug, debug|release) {
    DESTDIR = $$QT_ROOT/bind/plugins
} else {
    DESTDIR = $$QT_ROOT/bin/plugins
}

2. Plugin file (plugin.cpp)

#define QSCRIPT_PLUGIN_BUILD
#include "script_plugin_helper.h"

const QScriptHostAPI* g_api = nullptr;

// Simple function
QScriptValueHandle SayHello(QScriptContextHandle ctx,
                            const QScriptArg* args, int arg_count)
{
    CHECK_MIN_PARAMS(1, "SayHello");
    ScriptString name = getString(args[0].value);
    ScriptString msg = Q("Hello, ") + name + Q("!");
    return g_api->CreateString(ctx, msg.c_str());
}

// Initialization
QSCRIPT_EXPORT int QSCRIPT_API
QodeScriptInitialize(const QScriptHostAPI* host_api)
{
    if (!host_api || host_api->version < 1) return 0;
    g_api = host_api;
    return 1;
}

// Registering functions
QSCRIPT_EXPORT void QSCRIPT_API
QodeScriptRegisterFunctions(QScriptContextHandle ctx)
{
    g_api->RegisterFunction(ctx, Q("SayHello"), SayHello);
}

// Shutdown
QSCRIPT_EXPORT void QSCRIPT_API QodeScriptShutdown(void)
{
    g_api = nullptr;
}

// Export the interface
extern "C" {
QSCRIPT_EXPORT const QScriptPluginInterface* QSCRIPT_API
QodeScriptGetPlugin(void)
{
    static QScriptPluginInterface iface = {
        1, "MyPlugin", "Author", 1,
        QodeScriptInitialize,
        QodeScriptRegisterFunctions,
        QodeScriptShutdown
    };
    return &iface;
}

#ifdef _WIN32
int _libmain(unsigned long) { return 1; }
#endif
}

3. Usage in a script

loadPlugin("myplugin");
print(SayHello("World"));

Full example — plugin object

#define QSCRIPT_PLUGIN_BUILD
#include "script_plugin_helper.h"

const QScriptHostAPI* g_api = nullptr;

class Counter : public PluginObject
{
public:
    Counter() : count_(0)
    {
        // Property count (read-only)
        BIND_PROPERTY(this, "count",
            [this](QScriptContextHandle ctx) {
                return g_api->CreateInt(ctx, count_);
            }
        );

        // Method increment
        BIND_METHOD(this, "increment", &Counter::Increment);

        // Method reset
        BIND_METHOD(this, "reset", &Counter::Reset);
    }

    QScriptValueHandle Increment(QScriptContextHandle ctx,
                                  const QScriptArg* args, int arg_count)
    {
        CHECK_MIN_PARAMS(0, "increment");
        int64_t step = 1;
        if (arg_count >= 1)
            step = getLong(args[0].value);
        count_ += step;
        return g_api->CreateInt(ctx, count_);
    }

    QScriptValueHandle Reset(QScriptContextHandle ctx,
                              const QScriptArg* args, int arg_count)
    {
        CHECK_MIN_PARAMS(0, "reset");
        count_ = 0;
        return g_api->CreateNull(ctx);
    }

private:
    int64_t count_;
};

// Object factory
static QScriptValueHandle CreateCounter(QScriptContextHandle ctx,
                                         const QScriptArg* args, int arg_count)
{
    return CreatePluginObject(ctx, new Counter());
}

// Plugin export
QSCRIPT_EXPORT int QSCRIPT_API
QodeScriptInitialize(const QScriptHostAPI* host_api)
{
    if (!host_api || host_api->version < 1) return 0;
    g_api = host_api;
    return 1;
}

QSCRIPT_EXPORT void QSCRIPT_API
QodeScriptRegisterFunctions(QScriptContextHandle ctx)
{
    g_api->RegisterFunction(ctx, Q("Counter"), CreateCounter);
}

QSCRIPT_EXPORT void QSCRIPT_API QodeScriptShutdown(void)
{
    g_api = nullptr;
}

extern "C" {
QSCRIPT_EXPORT const QScriptPluginInterface* QSCRIPT_API
QodeScriptGetPlugin(void)
{
    static QScriptPluginInterface iface = {
        1, "Counter Plugin", "Author", 1,
        QodeScriptInitialize,
        QodeScriptRegisterFunctions,
        QodeScriptShutdown
    };
    return &iface;
}
#ifdef _WIN32
int _libmain(unsigned long) { return 1; }
#endif
}

Iterating over collections

Direct iteration (low-level)

void ProcessTuple(QScriptContextHandle ctx, QScriptValueHandle tuple)
{
    QScriptValueHandle key, value;
    void* iterator = nullptr;
    QScriptResult res;

    if (g_api->TupleFirst(ctx, tuple, &key, &value, &iterator, &res))
    {
        do {
            // Process key, value
            ScriptValueEx k(g_api, ctx, key);
            ScriptValueEx v(g_api, ctx, value);
            Logger::Debug("Key: " + Utf16ToUtf8(k.toString(ctx)));

        } while (g_api->TupleNext(ctx, tuple, iterator, &key, &value, &res));
    }
    // the iterator is destroyed automatically when the end is reached
    // or early via TupleIteratorDestroy(iter)
}

Range-based for (C++ helper)

void ProcessTuple(QScriptContextHandle ctx, QScriptValueHandle tuple)
{
    for (auto& [key, value] : TupleRange(g_api, ctx, tuple))
    {
        ScriptValueEx k(g_api, ctx, key);
        ScriptValueEx v(g_api, ctx, value);
        // processing...
    }
}

Keeping values beyond the iteration

ScriptValue persistent_key;
for (auto& [k, v] : TupleRange(g_api, ctx, tuple))
{
    persistent_key = ScriptValue(g_api, ctx, k);
    // persistent_key is valid after the loop
}

Inter-plugin interaction (services)

Plugins can provide methods to be called by other plugins.

Registering a service

static QScriptValueHandle MyServiceFunc(
    QScriptContextHandle ctx, const QScriptArg* args,
    int arg_count, void* user_data)
{
    // Service implementation
    return g_api->CreateInt(ctx, 42);
}

g_api->RegisterService(ctx,
    Q("MyService"),     // service name
    Q("Process"),       // method name
    MyServiceFunc,      // function
    1,                  // minimum number of arguments
    -1,                 // maximum (-1 = unlimited)
    nullptr,            // user_data
    g_pluginId          // plugin handle
);

Calling a service from another plugin

QScriptResult res;
QScriptValueHandle result = g_api->CallService(ctx,
    Q("MyService"), Q("Process"), args, arg_count, &res);

Common mistakes and how to avoid them

1. Storing QScriptValueHandle

// WRONG:
QScriptValueHandle saved_handle;
void Func(QScriptContextHandle ctx, QScriptValueHandle h) {
    saved_handle = h;  // h will be invalid after returning!
}

// RIGHT:
ScriptValue saved_ref;
void Func(QScriptContextHandle ctx, QScriptValueHandle h) {
    saved_ref = ScriptValue(g_api, ctx, h);  // persistent ref
}

2. Memory leaks with iterators

// WRONG — leaks if break before the end:
while (...) {
    if (cond) break;  // iterator not destroyed!
}

// RIGHT — use ScriptStlIterator or call Destroy:
void* iter;
g_api->TupleFirst(ctx, h, &key, &val, &iter, &res);
// ... on break:
g_api->TupleIteratorDestroy(iter);

3. Wrong context for handle(ctx)

// WRONG:
QScriptValueHandle h = some_ref_to_handle;
// ... later, in another context:
ScriptValueEx val(g_api, other_ctx, h);  // h may be invalid!

// RIGHT — always use the same ctx:
ScriptValueEx val(g_api, ctx, ref.handle(ctx));

4. ArenaLevelScope — critical error

// WRONG — the value is destroyed when leaving the scope:
QScriptValueHandle HostGet(QScriptContextHandle ctx, QScriptValueHandle obj)
{
    ArenaLevelScope scope{ctx->arena};
    auto val = MakeTempValue(ctx, ScriptValue(int64_t(42)));
    return val;  // val points to destroyed memory!
}

// RIGHT — copy before the scope ends:
QScriptValueHandle HostGet(QScriptContextHandle ctx, QScriptValueHandle obj)
{
    ArenaLevelScope scope{ctx->arena};
    auto val = MakeTempValue(ctx, ScriptValue(int64_t(42)));
    return val->Clone();
}

See Arena safety for details.

5. Forgotten argument check

// WRONG:
QScriptValueHandle MyFunc(QScriptContextHandle ctx,
                           const QScriptArg* args, int arg_count)
{
    ScriptString s = getString(args[0].value);  // crash if arg_count == 0
}

// RIGHT:
QScriptValueHandle MyFunc(QScriptContextHandle ctx,
                           const QScriptArg* args, int arg_count)
{
    CHECK_MIN_PARAMS(1, "MyFunc");
    ScriptString s = getString(args[0].value);
}
