Archive 7-Zip Plugin

Working with archives — 7z, ZIP, TAR, GZIP via 7-Zip

RUEN

Home / Plugins / Archive 7-Zip

Dependencies

Requires an installed 7z.exe (or 7z on Linux). By default the plugin looks for 7z.exe in PATH.

Loading the plugin

loadPlugin("plugins/archive7zip.dll");

Creating the object

// With the default path (7z.exe in PATH)
var zip = archive7zip();

// With an explicit path to 7z.exe
var zip = archive7zip("C:\\Program Files\\7-Zip\\7z.exe");

list(archive, [password])

Returns the list of files in the archive with metadata.

Tuple fieldTypeDescription
pathstringFile path inside the archive
sizeintSize in bytes
modifiedstringModification date
crcstringCRC checksum
encryptedint (0/1)Whether the file is encrypted
var zip = archive7zip();
var files = zip.list("archive.zip");
for (var f : files) {
    print(f.path + " — " + f.size + " bytes, CRC=" + f.crc);
}

// With a password
var files = zip.list("encrypted.7z", "mypassword");

extract(archive, destPath, [password], [withPaths])

Extracts the whole archive. Returns a tuple with the result.

FieldTypeDescription
successint (0/1)Whether the operation succeeded
exitCodeint7z return code
outputstring7z output
ArgumentTypeDefaultDescription
archivestring—Path to the archive
destPathstring—Where to extract to
passwordstring—Password
withPathsbooltrueKeep the directory structure (x vs e)
var zip = archive7zip();

// Extract keeping paths
var result = zip.extract("archive.zip", "C:\\output");
print(result.success);  // 1
print(result.output);   // 7z output

// Extract without the directory structure
zip.extract("archive.zip", "C:\\flat", "", false);

// With a password
zip.extract("encrypted.7z", "C:\\out", "mypassword");

extractFile(archive, fileName, destPath, [password])

Extracts one specific file from the archive.

var zip = archive7zip();
var ok = zip.extractFile("archive.zip", "docs/readme.txt", "C:\\output");
print(ok);  // 1

// With a password
zip.extractFile("encrypted.7z", "secret.txt", "C:\\out", "pass123");

create(archive, files, [format], [password], [level])

Creates a new archive from a list of files.

ArgumentTypeDefaultDescription
archivestring—Path to the archive being created
filesArray<string>—List of files to archive
formatstring"zip"Format: "zip", "7z", "tar", "gzip"
passwordstring—Password (enables encryption)
levelint5Compression level 0-9 (0=no compression, 9=maximum)
var zip = archive7zip();

// Simple ZIP
zip.create("backup.zip", #[
    "C:\\Docs\\file1.txt",
    "C:\\Docs\\file2.txt",
    "C:\\Photos\\image.jpg"
]);

// 7z with maximum compression
zip.create("archive.7z", #[
    "data1.bin",
    "data2.bin"
], "7z", "", 9);

// Encrypted archive
zip.create("secret.7z", #[
    "private.txt",
    "keys.dat"
], "7z", "StrongPassword123", 9);

// GZIP (one file — format limitation)
zip.create("data.tar.gz", #"data.tar", "gzip");

Limitations

  • GZIP: the format supports only one file. Passing several files will raise an error.
  • -mhe=on (metadata encryption): works only for the 7z format, not for ZIP.
  • Listing password-protected archives: 7z can show the file list of a password-protected ZIP archive even without the password (metadata is not encrypted). For 7z archives a password is required.

test(archive, [password])

Checks the integrity of the archive (7z t command).

var zip = archive7zip();
var ok = zip.test("backup.zip");
if (ok) {
    print("Archive is intact!");
} else {
    print("Archive is corrupted!");
}

// With a password
zip.test("encrypted.7z", "password");

info(archive)

Gets information about the archive. Returns a tuple {type} or null.

var zip = archive7zip();
var info = zip.info("archive.zip");
print(info.type);  // "zip"

info = zip.info("backup.7z");
print(info.type);  // "7z"

Examples

File backup

loadPlugin("plugins/archive7zip.dll");

var zip = archive7zip();
var today = datetime.now().strftime("%Y%m%d");
var archiveName = "backup_" + today + ".7z";

var files = file.listdir("C:\\Important");
zip.create(archiveName, files, "7z", "", 9);
print("Created: " + archiveName);

Extracting an archive

var zip = archive7zip();
var files = zip.list("download.zip");
print("Files in archive: " + files.size());

var result = zip.extract("download.zip", "C:\\Downloads\\extracted");
if (result.success) {
    print("Extracted successfully");
} else {
    print("Error: " + result.output);
}

Extracting a single file

var zip = archive7zip();
zip.extractFile("project.7z", "src/main.cpp", "C:\\MyProject");

Integrity check and info

var zip = archive7zip();
var info = zip.info("data.7z");
print("Format: " + info.type);

if (zip.test("data.7z")) {
    print("Archive is intact");
} else {
    print("Archive is corrupted!");
}

Custom path to 7z

var zip = archive7zip("D:\\Tools\\7-Zip\\7z.exe");
zip.tempPath = "D:\\Temp";
var files = zip.list("archive.zip");
