Quick start

// Hello world
print("Hello, world!");

// Variables and arithmetic
var x = 10;
var y = 3.14;
print("x = \(x), y = \(y)");

// Functions
function factorial(n) {
    if (n <= 1) return 1;
    return n * factorial(n - 1);
}

// Classes
class Point {
    var x;
    var y;
    Point(x, y) { this.x = x; this.y = y; }
    function distance(other) {
        return math.sqrt((this.x - other.x)**2 + (this.y - other.y)**2);
    }
}

// Collections
var arr = #[1, 2, 3, 4, 5];
var doubled = arr.mapped(function(x) { return x * 2; });
print(doubled); // [2, 4, 6, 8, 10]

// Modules
uses file;
file.write("out.txt", "Data");
