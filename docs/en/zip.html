Zip plugin

Working with ZIP archives without external programs (native implementation based on miniz)

RUEN

Home / Plugins / Zip

Features

  • No 7-Zip installation or external process calls required.
  • UTF-8 file name support (including Cyrillic).
  • Path traversal protection on extraction.
  • Password (ZIP encryption) is not supported.

Loading the plugin

loadPlugin("plugins/zip.dll");

Creating an object

var z = zip();

list(archive)

Returns an array of tuples (or null if the archive could not be opened).

Tuple fieldTypeDescription
pathstringFile path inside the archive
sizeintSize (uncompressed) in bytes
compressedintCompressed size in bytes
modifiedstringModification date (YYYY-MM-DD HH:MM:SS)
crcstringCRC32 checksum (hex)
methodstringCompression method: store, deflate
encryptedint (0/1)Whether the entry is encrypted
directoryint (0/1)Whether the entry is a directory
var z = zip();
var files = z.list("archive.zip");
for (var f : files) {
    print(f.path + " — " + f.size + " bytes, CRC=" + f.crc + ", " + f.method);
}

extract(archive, destPath, [password], [withPaths])

Extracts the archive into destPath. withPaths=true (default) keeps the directory structure, false extracts all files "flat" into the root. Passwords are not supported (encrypted archives cannot be extracted).

var z = zip();

// Keeping paths
var ok = z.extract("archive.zip", "C:\\output");

// Without directory structure
z.extract("archive.zip", "C:\\flat", "", false);

extractFile(archive, fileName, destPath, [password])

Extracts a single file by its name inside the archive.

var z = zip();
var ok = z.extractFile("archive.zip", "docs/readme.txt", "C:\\output");
print(ok);  // 1

create(archive, files, [format], [password], [level])

ArgumentTypeDefaultDescription
archivestring—Path to the archive to be created
filesArray<string>—Files or folders (folders are added recursively)
formatstring"zip"Only "zip" is supported
passwordstring—Encryption is not supported
levelint5Compression level 0-9 (0=no compression)
var z = zip();

// Simple ZIP from files
z.create("backup.zip", #[
    "C:\\Docs\\file1.txt",
    "C:\\Docs\\file2.txt"
]);

// A whole folder — recursively
z.create("data.zip", #["C:\\Data"]);

// No compression
z.create("store.zip", #["file.bin"], "zip", "", 0);

test(archive)

Checks the archive's integrity (unpacking all entries with CRC32 verification).

var z = zip();
if (z.test("backup.zip")) {
    print("Archive is intact!");
} else {
    print("Archive is corrupted!");
}

info(archive)

Returns tuple {type, files, size, compressed} or null.

var z = zip();
var info = z.info("archive.zip");
print(info.type);       // "zip"
print(info.files);      // number of entries
print(info.size);       // total uncompressed size

Example

loadPlugin("plugins/zip.dll");

var z = zip();
z.create("backup.zip", #["C:\\Important"]);

var files = z.list("backup.zip");
print("Files in archive: " + files.size());

if (z.test("backup.zip")) {
    z.extract("backup.zip", "C:\\restore");
    print("Restored successfully");
}
