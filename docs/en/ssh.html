SSH module

Built-in module for remote server management — SSH connections, command execution, file transfer via SCP and SFTP

RUEN

Home / SSH

Dependencies

Built into the engine. Uses libssh2 1.11.1 + OpenSSL 3.

To run on Windows the following must be next to the binary: libcrypto-3-x64.dll, libssl-3-x64.dll.

Creating a session

// ssh — global module object
var session = ssh.connect();

sftp_readdir result format

Each array element is a tuple:

#{
    "name":  "file.txt",   // string — file name
    "size":  12345,        // int    — size in bytes
    "mode":  33188,        // int    — POSIX permissions (octal 0644)
    "mtime": 1694000000,   // int    — modification time (unix timestamp)
    "isdir":  0            // int    — 1 if a directory
}

Example: connecting and executing a command

var session = ssh.connect();

if (session.connect("192.168.1.1", 22, "root", "password")) {
    print("Connected!");

    var result = session.exec("uname -a");
    print(result);

    var hostname = session.exec("hostname");
    print("Server: \(hostname)");

    session.disconnect();
} else {
    print("Error: \(session.last_error())");
}

Example: uploading a file via SCP

var session = ssh.connect();
session.connect("192.168.1.1", 22, "user", "pass");

// Downloading a remote file
var data = session.scp_recv("/var/log/syslog");
print("Size: \(size(data))");

// Sending data
session.scp_send("/tmp/upload.txt", "Hello from Script!");

session.disconnect();

Example: working with SFTP

var s = ssh.connect();
s.connect("192.168.1.1", 22, "user", "pass");

// SFTP initialization
s.sftp_init();

// Browsing a directory
var files = s.sftp_readdir("/home/user");
for (var f : files) {
    print("\(f.name)  \(f.size) bytes");
}

// Writing a file
var h = s.sftp_open("/tmp/test.txt", "w");
s.sftp_write(h, "Data from the script\n");
s.sftp_close(h);

// Reading a file
var r = s.sftp_open("/tmp/test.txt", "r");
var content = s.sftp_read(r, 4096);
s.sftp_close(r);
print(content);

// File information
var st = s.sftp_stat("/tmp/test.txt");
print("Size: \(st.size)");

// File operations
s.sftp_rename("/tmp/test.txt", "/tmp/renamed.txt");
s.sftp_unlink("/tmp/renamed.txt");

// Directories
s.sftp_mkdir("/tmp/newdir");
s.sftp_rmdir("/tmp/newdir");

s.disconnect();

Example: bulk command execution

var servers = [
    #{"host": "192.168.1.10", "user": "admin", "pass": "secret"},
    #{"host": "192.168.1.11", "user": "admin", "pass": "secret"},
    #{"host": "192.168.1.12", "user": "admin", "pass": "secret"}
];

for (var srv : servers) {
    var s = ssh.connect();
    if (s.connect(srv.host, 22, srv.user, srv.pass)) {
        print("\(srv.host):");
        print(s.exec("uptime"));
        s.disconnect();
    } else {
        print("\(srv.host): FAILED");
    }
}

Example: disk monitoring

var s = ssh.connect();
s.connect("server.example.com", 22, "root", "pass");

// Checking free space
var df = s.exec("df -h /");
print("Disk space:");
print(df);

// Checking processes
var ps = s.exec("ps aux | head -20");
print("\nTop processes:");
print(ps);

// Checking directory size
var du = s.exec("du -sh /var/log");
print("\nLogs:");
print(du);

s.disconnect();

Example: fetching a configuration

var s = ssh.connect();
s.connect("db-server", 22, "dba", "password");
s.sftp_init();

// Reading the configuration file
var h = s.sftp_open("/etc/my.cnf", "r");
var config = s.sftp_read(h, 65536);
s.sftp_close(h);

// Searching for the needed lines
var lines = config.splitlines();
for (var line : lines) {
    if (line.startswith("max_connections")) {
        print("Found: \(line)");
    }
}

s.disconnect();
