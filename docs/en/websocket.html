WebSocket Plugin

WebSocket client — bidirectional real-time communication, text and binary messages, ping/pong

RUEN

Home / Plugins / WebSocket

Dependencies

Requires OpenSSL 3: libssl-3-x64.dll, libcrypto-3-x64.dll (placed next to the binary).

Incoming text and binary messages are reassembled from WebSocket fragments; the total message size is limited to 64 MiB.

Loading the plugin

loadPlugin("plugins/websocket.dll");

Creating the object

var ws = WebSocket();

Connecting

Supports ws:// (plain) and wss:// (SSL/TLS).

var ws = WebSocket();

// ws://
ws.connect("ws://echo.websocket.events");

// wss://
ws.connect("wss://echo.websocket.events");

// with a port
ws.connect("ws://localhost:8080");

// with a path
ws.connect("wss://example.com/chat");

print(ws.isconnected());  // 1

Text messages

var ws = WebSocket();
ws.connect("wss://echo.websocket.events");

// Sending
ws.send("Hello, world!");
ws.send("val=\(42)");

// Receiving
var msg = ws.recv();
if (msg != null) {
    print("Received: \(msg)");
}

// Checking for messages
if (ws.hasMessage()) {
    var next = ws.recv();
}

Binary messages

var ws = WebSocket();
ws.connect("wss://echo.websocket.events");

// Sending binary data (ByteArray)
var ba = ByteArray();
ba.push(0x48);  // H
ba.push(0x65);  // e
ba.push(0x6C);  // l
ba.push(0x6C);  // l
ba.push(0x6F);  // o
ws.sendBinary(ba);

// Receiving binary data
var data = ws.recvBinary();
if (data != null) {
    print("Size: \(data.size())");
    print("First byte: \(data[0])");
}

// Sending a string as binary data
ws.sendBinary("raw bytes");

Ping / Pong

The plugin automatically replies to server ping frames with pong frames.

var ws = WebSocket();
ws.connect("wss://echo.websocket.events");

// Manual ping
ws.ping();

Closing the connection

var ws = WebSocket();
ws.connect("wss://echo.websocket.events");

// Graceful close (sends a close frame)
ws.close();

// Abrupt disconnect (no close frame)
ws.disconnect();

Message queue

var ws = WebSocket();
ws.connect("wss://echo.websocket.events");

// Number of messages in the queue
print(ws.messageCount());

// Whether there are messages
print(ws.hasMessage());  // 1 or 0

// Processing all messages
while (ws.hasMessage()) {
    var msg = ws.recv();
    print(msg);
}

// Clearing the queue
ws.clearMessages();

Timeout

var ws = WebSocket();
ws.connect("wss://echo.websocket.events");

// Setting the read timeout (ms)
ws.setTimeout(5000);  // 5 seconds

// Getting the current timeout
var timeout = ws.setTimeout();  // getter

Example: Echo client

loadPlugin("plugins/websocket.dll");

var ws = WebSocket();

if (ws.connect("wss://echo.websocket.events")) {
    print("Connected!");

    // Sending messages
    ws.send("Hi!");
    ws.send("Echo test");

    // Waiting for replies
    system.sleep(2000);

    while (ws.hasMessage()) {
        var msg = ws.recv();
        print("Echo: \(msg)");
    }

    ws.close();
} else {
    print("Error: \(ws.lastError())");
}

Example: Receiving binary data

loadPlugin("plugins/websocket.dll");

var ws = WebSocket();
ws.connect("wss://echo.websocket.events");

// Sending binary data
var data = ByteArray();
for (var i = 0; i < 256; i++) {
    data.push(i);
}
ws.sendBinary(data);

system.sleep(1000);

var received = ws.recvBinary();
if (received != null) {
    print("Bytes received: \(received.size())");
}

ws.close();

Example: Error handling

loadPlugin("plugins/websocket.dll");

var ws = WebSocket();
ws.setTimeout(5000);

if (!ws.connect("wss://invalid-host.example.com")) {
    print("Connection error: \(ws.lastError())");
    return;
}

ws.send("test");

if (ws.recv() == null) {
    print("No response. Error: \(ws.lastError())");
}

ws.close();

Example: Without SSL (ws:// only)

Connecting to servers without encryption:

var ws = WebSocket();
ws.connect("ws://localhost:9001");

Connecting to servers with SSL:

var ws = WebSocket();
ws.connect("wss://secure-server.com");
