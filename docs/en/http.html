Dependencies

get(url)

Performs a GET request. Returns true if a response was received (any status).

ArgumentTypeDescription
urlstringFull URL (http://... or https://...)
var http = HttpClient();
if (http.get("https://httpbin.org/get")) {
    print(http.status());        // 200
    print(http.statusText());    // OK
    print(http.getResponse());   // response body
}

post(url, [body], [contentType])

POST request with a body. Default Content-Type: application/x-www-form-urlencoded.

ArgumentTypeDefaultDescription
urlstring—Full URL
bodystring""Request body
contentTypestring"application/x-www-form-urlencoded"Content-Type
// Form data
http.post("https://httpbin.org/post", "name=Ivan&age=30");

// JSON
http.post("https://httpbin.org/post",
    "{\"key\":\"value\"}", "application/json");

put(url, [body], [contentType])

PUT request. Similar to post().

http.put("https://httpbin.org/put", "updated data", "text/plain");

del(url)

DELETE request.

http.del("https://httpbin.org/delete");

patch(url, [body], [contentType])

PATCH request. Similar to post().

http.patch("https://httpbin.org/patch", "partial update");

head(url)

HEAD request — returns only headers without a body.

http.head("https://httpbin.org/get");
print(http.getHeader("Content-Type"));  // application/json

request(method, url, [body], [contentType])

Arbitrary HTTP method (OPTIONS, TRACE, CONNECT, etc.).

http.request("OPTIONS", "https://httpbin.org/get");

download(url, filePath)

Download a file to disk. Supports HTTP and HTTPS with automatic redirection.

ArgumentTypeDescription
urlstringFile URL
filePathstringPath to save to
if (http.download("https://example.com/file.zip", "download/file.zip")) {
    print("Size: " + file.size("download/file.zip"));
}

upload(url, filePath, [fieldName])

Multipart/form-data file upload (equivalent to <input type="file">).

ArgumentTypeDefaultDescription
urlstring—Upload URL
filePathstring—Local path to the file
fieldNamestring"file"Field name in the form
http.upload("https://httpbin.org/post", "photo.jpg");
http.upload("https://httpbin.org/post", "data.csv", "attachment");

Headers and authentication

// Custom headers
http.setHeader("X-Custom", "value");
http.setHeader("Accept", "application/json");
http.get("https://httpbin.org/headers");
http.clearHeaders();  // + resets basic/bearer auth

// Basic Auth
http.setBasicAuth("admin", "secret");
http.get("https://httpbin.org/basic-auth/admin/secret");
http.clearHeaders();

// Bearer Token
http.setBearerAuth("eyJhbGciOiJIUzI1NiJ9...");
http.get("https://api.example.com/me");
http.clearHeaders();

// Custom User-Agent
http.setUserAgent("MyApp/2.0");

Timeout and connection

// Timeout (default 30000 ms)
http.setTimeout(5000);   // set
print(http.setTimeout()); // read current

// Manual connection handling
http.connect("api.example.com", 443, true);
print(http.isconnected());  // true
http.disconnect();

Redirects

By default the plugin automatically follows HTTP redirects (301, 302, 303, 307, 308) up to 10 times.

// Disable redirects
http.setFollowRedirects(false);
http.get("https://httpbin.org/redirect/3");
print(http.status());  // 302

// Enable them back
http.setFollowRedirects(true);

Response and headers

http.get("https://httpbin.org/get");

// Status
print(http.status());       // 200
print(http.statusText());   // OK

// Response body
var text = http.getResponse();     // string
var data = http.getResponseBytes(); // ByteArray

// Response headers
print(http.getHeader("Content-Type"));
var all = http.getResponseHeaders(); // tuple

Utilities

// URL encoding/decoding
var enc = http.urlEncode("Hello World!");  // %48%65%6C%6C%6F...
var dec = http.urlDecode("%D0%9F%D1%80%D0%B8%D0%B2%D0%B5%D1%82");

// MIME type by extension
print(http.getMimeType("index.html"));   // text/html
print(http.getMimeType("photo.jpg"));    // image/jpeg
print(http.getMimeType("data.json"));    // application/json
print(http.getMimeType("app.exe"));      // application/x-msdownload

// Diagnostics
print(http.lastError());  // "null" if there are no errors

Examples

GET request with error handling

loadPlugin("plugins/http.dll");

var http = HttpClient();
http.setTimeout(10000);

if (http.get("https://httpbin.org/get")) {
    if (http.status() == 200) {
        print(http.getResponse());
    } else {
        print("HTTP Error: " + http.status() + " " + http.statusText());
    }
} else {
    print("Connection error: " + http.lastError());
}

POST JSON

var http = HttpClient();
var json = "{\"name\":\"Ivan\",\"age\":30}";

http.post("https://httpbin.org/post", json, "application/json");
print(http.status());        // 200
print(http.getResponse());   // echo from the server

File download

var http = HttpClient();
if (http.download("https://httpbin.org/robots.txt", "robots.txt")) {
    print("Downloaded: " + file.size("robots.txt") + " bytes");
}

// Multipart upload
var http2 = HttpClient();
http2.upload("https://httpbin.org/post", "photo.jpg", "photo");
print(http2.getResponse());

Working with an API (JSON + headers)

var http = HttpClient();
http.setHeader("Accept", "application/json");
http.setBearerAuth("my-api-token");
http.setUserAgent("MyScript/1.0");

if (http.get("https://api.example.com/users")) {
    var data = json.parse(http.getResponse());
    print(data);
}

http.clearHeaders();

Proxy

var http = HttpClient();
http.setProxy("proxy.company.com", 8080);
http.get("https://httpbin.org/ip");
print(http.getResponse());

HttpServer — built-in HTTP server

Multithreaded HTTP server with a callback request handling model. It accepts connections in a background thread; the script handles requests synchronously via process().

Creating an object

var server = HttpServer();

Server methods

MethodArgumentsReturnsDescription
listenportboolStart listening on a port
stop—boolStop the server
islistening—boolCheck listening state
port—intCurrent port number
process[timeout_ms]tuple|nullProcess one incoming request (returns the request tuple or null)
respondsocketId, responseTupleboolSend an HTTP response and close the connection
lastError—stringLast error

listen(port)

Starts listening on the specified TCP port. Spawns a background thread to accept connections. Returns true on success.

var server = HttpServer();
if (server.listen(8080)) {
    print("Server started on port " + server.port());
}

process(timeout_ms)

Waits for an incoming request for timeout_ms milliseconds (default 50). If a request arrived — returns a tuple with the request data. If there are no requests — returns null.

The request tuple contains:

FieldTypeDescription
methodstringHTTP method (GET, POST, ...)
pathstringRequest path (/api/data)
querystringQuery string (q=test&page=1)
bodystringRequest body
headerstupleRequest headers (key: value)
portintServer port
socketintConnection identifier (to pass to respond)

respond(socketId, responseTuple)

Sends an HTTP response to the client and closes the connection. responseTuple is a tuple with the fields:

FieldTypeDefaultDescription
statusint200HTTP status
statusTextstring"OK"Status text
bodystring""Response body
headerstuple#{} Response headers

Example: simple server

loadPlugin("plugins/http.dll");

var server = HttpServer();
server.listen(8080);
print("Server started on port 8080");

var handler = function(req) {
    print(req.method + " " + req.path);
    return #{
        "status": 200,
        "statusText": "OK",
        "headers": #{"Content-Type": "text/plain"},
        "body": "Hello, World!"
    };
};

// Main processing loop
while (server.islistening()) {
    var req = server.process(100);
    if (req != null) {
        var resp = handler(req);
        server.respond(req.socket, resp);
    }
}

Example: standalone server script

// server.prg — run: qodeqt.exe server.prg
loadPlugin("plugins/http.dll");

var server = HttpServer();
server.listen(3000);

var handler = function(req) {
    if (req.method == "GET" && req.path == "/time") {
        return #{
            "status": 200,
            "headers": #{"Content-Type": "application/json"},
            "body": "{\"time\": \"\(datetime.now())\"}"
        };
    }
    return #{
        "status": 404,
        "statusText": "Not Found",
        "body": "Not Found"
    };
};

while (server.islistening()) {
    var req = server.process(100);
    if (req != null) {
        var resp = handler(req);
        server.respond(req.socket, resp);
    }
}
