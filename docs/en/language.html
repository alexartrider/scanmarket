Literals

"Simple string"

"Concatenation: "   "two lines"    // result: "Concatenation: two lines"
"Multi-line\
string"

Line continuation via \ at the end of a line, or concatenation of adjacent literals.

Interpolation \(expression) substitutes the value of an expression; arbitrary expressions are allowed inside, including nested string literals and balanced parentheses: "\(coalesce(name, "none"))", "\(str.format("%.2f", x))".

Export

The export keyword is placed after the declaration (not before it).

function helper() export { ... }     // function is visible outside the module
var x = 0 export;                    // variable is visible outside the module
var y export;                        // exported without initialization

Classes

class Person {
    var name;
    var age;

    // Constructor
    Person(name, age) {
        this.name = name;
        this.age = age;
    }

    // Method
    function greet() {
        return "I am \(this.name), \(this.age) years old";
    }

    // Destructor
    ~Person() { }
}

// Inheritance
class Student : Person {
    var university;

    Student(name, age, university) {
        this::Person(name, age);  // base constructor call
        this.university = university;
    }
}

var s = Student("Ivan", 20, "MSU");
print(s.greet());          // I am Ivan, 20 years old
print(s.university);       // MSU

Examples from sample/

Classes and Inheritance

class Animal {
    Animal(name, sound) {
        this.name = name;
        this.sound = sound;
    }
    function speak() {
        return "\(this.name) says \(this.sound)!";
    }
}

class Dog : Animal {
    Dog(name) : Animal(name, "Woof") {   // base constructor call
        this.tricks = 0;
    }
    function fetch() {
        this.tricks = this.tricks + 1;
        return "\(this.name) fetches the stick! (trick #\(this.tricks))";
    }
}

var rex = Dog("Rex");
print(rex.speak());       // Rex says Woof!
print(rex.fetch());       // Rex fetches the stick! (trick #1)

Collections — Methods

Array — Basics

var arr = #[3, 1, 4, 1, 5];
arr.size();                   // 5 — number of elements
arr.empty();                  // false
arr.sort();                   // sort (mutates)
arr.reverse();                // reverse (mutates)
arr.clear();                  // clear the array
arr.reserve(100);             // reserve memory
arr.add(6);                   // append an element
arr.push(7, 8);               // append several elements
arr.pop();                    // remove and return the last element
arr.insert(0, 99);            // insert at index
arr.removeAt(0);              // remove at index
arr.remove(4);                // remove all occurrences
arr.removeAll(1);             // remove all occurrences
arr.swap(0, 2);               // swap elements
arr.clone();                  // deep copy

Array — Search and Checks

arr.contains(4);               // true — whether it contains the element
arr.indexOf(4);                // index of first occurrence, -1 if not found
arr.lastIndexOf(1);            // index of last occurrence
arr.count(1);                  // number of occurrences
arr.min();                     // minimum element
arr.max();                     // maximum element
arr.sum();                     // sum of elements
arr.average();                 // average value

Array — Slices

arr.sliced(0, 3);              // new array [0..3)
arr.mid(1);                    // slice from position to end
arr.mid(1, 3);                 // slice [pos, pos+count)
arr.first();                    // first element
arr.last();                     // last element

Array — Sorting

arr.sort();                    // default
arr.sort(function(a, b) { return a - b; });  // custom sort
arr.reversed();                // new sorted array (does not mutate)

Array — Functional

var doubled = map(arr, function(x) { return x * 2; });
var evens = filter(arr, function(x) { return x % 2 == 0; });

arr.some(function(x) { return x > 3; });    // whether at least one matches
arr.every(function(x) { return x > 0; });  // whether all match

Array — Grouping

var words = #["apple", "banana", "avocado", "blueberry"];
var groups = words.grouped(function(w) { return w.first(); });
// #{"a": #["apple","avocado"], "b": #["banana","blueberry"]}

var nums = #[1, 2, 3, 4, 5, 6];
var groups2 = nums.grouped(function(n) { return n % 2 == 0 ? "even" : "odd"; });
// #{"odd": #[1,3,5], "even": #[2,4,6]}

String Methods

Search

"Hello World".contains("World")       // true
"Hello World".pos("World")            // 6 — substring index, -1 if not found
"Hello World".rfind("l")              // 9 — search from the end
"Hello World".rfind("l", 8)            // 3 — rfind from position
"hello".find_first_of("aeiou")         // 1 — first character from the set
"hello".find_last_of("aeiou")          // 4 — last character from the set
"hello".find_first_not_of("h")        // 1 — first character not in the set
"hello".find_last_not_of("o")         // 3 — last character not in the set

Checks

"Hello World".startswith("Hello")     // true
"Hello World".endswith("World")       // true
"hello".comparenocase("HELLO")         // 0 — case-insensitive comparison
"hello".comparenocase("world")         // -1

Case

"Hello World".upper()                 // "HELLO WORLD"
"Hello World".lower()                 // "hello world"
"Hello World".swapcase()              // "hELLO wORLD"

Formatting

"hi".center(10)                      // "    hi    "
"hi".center(10, "*")                  // "****hi****"
"hi".ljust(10)                       // "hi        "
"hi".ljust(10, "*")                   // "hi********"
"hi".rjust(10)                       // "        hi"
"hi".rjust(10, "*")                   // "********hi"
"42".zfill(6)                         // "000042"
"hi".pad(10, "-")                     // "hi--------"

Substring Extraction

"hello".sub(0, 3)                    // "hel" — substring [pos, pos+count)
"hello".left(3)                      // "hel"
"hello".right(3)                     // "llo"
"hello".mid(1)                       // "ello" — from position to end
"hello".first(3)                     // "hel" — first N characters
"hello".first()                      // "h" — first character
"hello".last(3)                      // "llo" — last N characters
"hello".last()                       // "o" — last character
"hello world".chopped(3)             // "hello wor" — without the last N characters
"hello world".chopped(1)             // "hello worl"

Trimming and Normalization

"  Hello  ".trim()                   // "Hello"
"  Hello  ".trimleft()               // "Hello  "
"  Hello  ".trimright()              // "  Hello"
"  hello   world  ".simplified()     // "hello world" — whitespace normalization
'"hello"'.stripquotes()               // "hello" — removes outer quotes
"hello".quoted()                     // "&quot;hello&quot;" — wraps in quotes
"dir/sub/".trimtrailingslashes()     // "dir/sub" — removes trailing /

Splitting and Joining

"a,b,c".split(",")                    // #["a", "b", "c"]
#["a","b","c"].join("-")              // "a-b-c"
#["a","b","c"].joined("-")            // "a-b-c"

Regular Expressions

Patterns use ECMAScript std::regex. Strings are matched as UTF-8 bytes, not Unicode characters; \w, dot, and character boundaries are not Unicode-aware. String regex methods may throw for an invalid pattern.

"12345".regexmatch("\\d+")            // true — the whole string matches
"hello 123".regexsearch("\\d+")       // true — partial match
"hello 123 world".regexreplace("\\d+", "NUM")  // "hello NUM world"
"a,b,,c".regexsplit(",").size()       // 4
"a1 b2 c3".regexfindall("\\d")       // #["1","2","3"]

Unicode

String-search indices and positions count UTF-16 code units; a non-BMP code point occupies two positions.

"hello".codepointcount()             // 5 — number of codepoints
"ABC".codepointat(0)                  // 65 — Unicode code of the character
"hello".isvalidutf16()               // true
"ABC".tocodepoints()                  // #[65, 66, 67]

Utilities

"abc 123 def 456".extractnumbers()   // #[123, 456]
"Hello World".len()                   // 11
"Hello".repeat(3)                     // "HelloHelloHello"
"hello".replace("l", "r")              // "herro"
"a,b,c".toArray()                    // #["a","b","c"]
"a,b,c".tobytes()                    // ByteArray
var n = "123".tonumber()              // 123

Method Chaining

"  Hello World  ".trim().upper()                          // "HELLO WORLD"
"a,b,c".split(",").reverse().joined("-")                  // "c-b-a"
"hello world".simplified().upper()                        // "HELLO WORLD"
