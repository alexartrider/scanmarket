Loading the Plugin

Two ways to use it:

// Method 1: via the object
if (!loadPlugin("plugins/crypto.dll")) { system.exit(1); }
var c = Crypto();
var hash = c.sha256("Hello");

// Method 2: via the global service
if (!loadPlugin("plugins/crypto.dll")) { system.exit(1); }
var hash = crypto.sha256("Hello");

Encoding and Decoding

MethodArgumentsReturnsDescription
base64Encodedata, [urlSafe]stringBase64 encoding. urlSafe=true — URL-safe alphabet (-_ instead of +/)
base64Decodedata, [urlSafe], [asBytes]string / bytesBase64 decoding
hexEncodedatastringEncode to a hex string (0-9a-f)
hexDecodedata, [asBytes]bytes / stringDecode a hex string
// Base64
var enc = crypto.base64Encode("Hello world");
print(enc);  // "SGVsbG8gd29ybGQ="

var dec = crypto.base64Decode(enc);
print(dec);  // "Hello world"

// URL-safe Base64
var url = crypto.base64Encode("data/signature", true);

// Hex
var hex = crypto.hexEncode("Hello");
print(hex);  // "48656c6c6f"

var raw = crypto.hexDecode("48656c6c6f");
print(raw);  // "Hello"

Hash Functions

MethodArgumentsReturnsDescription
md5data, [asHex]string / bytesMD5 (128 bits)
sha1data, [asHex]string / bytesSHA-1 (160 bits)
sha256data, [asHex]string / bytesSHA-256 (256 bits)
sha512data, [asHex]string / bytesSHA-512 (512 bits)
sha3_256data, [asHex]string / bytesSHA3-256 (Keccak)
sha3_512data, [asHex]string / bytesSHA3-512 (Keccak)

By default they return a hex string. asHex=false — they return a ByteArray.

// Hash of a string
print(crypto.md5("Hello"));      // "8b1a9953c4611296a827abf8c47804d7"
print(crypto.sha256("Hello"));    // "185f8db32271fe25f561a6fc938b2e26..."

// Hash of a file
var data = file.readbytes("image.png");
print(crypto.sha256(data));

// As a ByteArray (not hex)
var hash = crypto.sha256("test", false);
print(hash.size());  // 32 bytes

HMAC and Signature Verification

MethodArgumentsReturnsDescription
hmackey, data, [algo], [asHex]string / bytesHMAC signature. algo: sha256 (default), sha512, sha1, md5
hmacVerifykey, data, signature, [algo]int (0/1)HMAC signature verification
pbkdf2password, salt, [iterations], [keyLen], [algo]bytesKey derivation. iterations=100000, keyLen=32, algo=sha256
// HMAC signature
var key = "secret_key";
var sig = crypto.hmac(key, "important data");
print(sig);  // hex string

// Signature verification
var ok = crypto.hmacVerify(key, "important data", sig);
print(ok);  // 1

// Invalid signature
ok = crypto.hmacVerify(key, "tampered data", sig);
print(ok);  // 0

// PBKDF2 — key derivation from a password
var salt = crypto.randomBytes(16);
var key1 = crypto.pbkdf2("password", salt, 100000, 32);
var key2 = crypto.pbkdf2("password", salt, 100000, 32);
print(key1.toString() == key2.toString());  // 1 (deterministic)

AES Encryption (GCM)

MethodArgumentsReturnsDescription
aesEncryptdata, key, [iv], [aad]bytesAES-GCM encryption. Format: IV(12) + Tag(16) + Ciphertext
aesDecryptdata, key, [aad], [asString]bytes / stringDecryption. asString=true — return a string

Keys that are not 16/24/32 bytes long are automatically hashed with SHA-256.

// Encryption
var key = "my_secret_key";
var encrypted = crypto.aesEncrypt("Secret message", key);
print(encrypted.size());  // IV(12) + Tag(16) + data

// Decryption
var decrypted = crypto.aesDecrypt(encrypted, key, "", true);
print(decrypted);  // "Secret message"

// With AAD (Additional Authenticated Data)
var aad = "header_context";
var enc = crypto.aesEncrypt("data", key, "", aad);
var dec = crypto.aesDecrypt(enc, key, aad, true);
print(dec);  // "data"

// Decryption with a wrong key — an error
var bad = crypto.aesDecrypt(enc, "wrong_key", aad);
print(bad.isnull());  // true

RSA

MethodArgumentsReturnsDescription
rsaGenerate[bits]tupleKey generation. bits: 2048 (default), 4096. Returns {private, public} PEM
rsaEncryptdata, publicKeybytesEncryption with the public key (PKCS1 OAEP)
rsaDecryptdata, privateKeystringDecryption with the private key
rsaSigndata, privateKey, [algo]bytesDigital signature. algo: sha256 (default)
rsaVerifydata, signature, publicKey, [algo]int (0/1)Signature verification
// Key generation
var keys = crypto.rsaGenerate();
print(keys.private);  // PEM of the private key
print(keys.public);   // PEM of the public key

// Encryption/decryption
var enc = crypto.rsaEncrypt("Secret message", keys.public);
var dec = crypto.rsaDecrypt(enc, keys.private);
print(dec);  // "Secret message"

// Digital signature
var data = "Important document";
var sig = crypto.rsaSign(data, keys.private);
print(crypto.rsaVerify(data, sig, keys.public));  // 1
print(crypto.rsaVerify("tampered", sig, keys.public));  // 0

MethodArgumentsReturnsDescription

// Key generation
var keys = crypto.ed25519Generate();

// Signing
var data = "Message to sign";

// Verification
print(crypto.ed25519Verify(data, sig, keys.public));  // 1
print(crypto.ed25519Verify("forged", sig, keys.public));  // 0

OTP (One-Time Passwords)

MethodArgumentsReturnsDescription
totpsecret, [digits], [period]stringTime-based OTP. digits=6, period=30 sec
hotpsecret, counter, [digits]stringHMAC-based OTP. counter — the counter

secret is a string in Base32 encoding.

// TOTP — one-time password
var secret = "JBSWY3DPEHPK3PXP";  // Base32
var code = crypto.totp(secret);
print(code);  // "123456" (6 digits)

// HOTP —.counter-based
var code1 = crypto.hotp(secret, 0);
var code2 = crypto.hotp(secret, 1);

JWT (JSON Web Token)

MethodArgumentsReturnsDescription
jwtCreatepayload, key, [algo], [header]stringJWT token generation
jwtVerifytoken, key, [algo]string / nullSignature verification + exp/nbf check. Returns the payload or null
jwtDecodetokentupleDecoding without verification. Returns {header, payload, signature}

Arguments:

  • payload — a JSON string or tuple (automatically serialized to JSON)
  • key — the secret (string/bytes) for HMAC, the PEM private key for RSA/ECDSA/EdDSA
  • algo — the signature algorithm (default HS256)
  • header — a tuple with additional header fields (optional)

Supported algorithms:

TypeAlgorithmsKey
HMACHS256, HS384, HS512Secret string or bytes
RSARS256, RS384, RS512PEM private key (creation) / PEM public key (verification)
ECDSAES256, ES384, ES512PEM private key (creation) / PEM public key (verification)
EdDSAEdDSAPEM private key (creation) / PEM public key (verification)

CLAIMS validation: during verification the following are checked automatically:

  • exp (expiration time) — the token is rejected if the current time is greater than exp
  • nbf (not before) — the token is rejected if the current time is less than nbf
// === HS256 (secret string) ===
var token = crypto.jwtCreate(
     '{"sub":"user1","name":"Ivan"}',
    "my_secret"
);
print(token);  // "eyJhbGciOiJI..."

// Verification
var payload = crypto.jwtVerify(token, "my_secret");
print(payload);  // '{"sub":"user1","name":"Ivan"}'

// Wrong key — null
print(crypto.jwtVerify(token, "wrong_key").isnull());  // true

// === Payload as a tuple (auto-serialization) ===
var token2 = crypto.jwtCreate(
    #{"sub": "user1", "role": "admin", "iat": 1700000000},
    "secret"
);

// === HS384 / HS512 ===
var token = crypto.jwtCreate("{}", "key", "HS512");

// === RS256 (RSA signature) ===
var keys = crypto.rsaGenerate();
var token = crypto.jwtCreate(
    '{"sub":"123"}',
    keys.private,
    "RS256"
);
var payload = crypto.jwtVerify(token, keys.public, "RS256");

// === ES256 (ECDSA signature) ===
// EC keys are supplied as ready-made PEM (no EC generation in the plugin):
// openssl ecparam -name prime256v1 -genkey -noout -out ec-private.pem
// openssl ec -in ec-private.pem -pubout -out ec-public.pem
var ecPrivateKey = file.readbytes("ec-private.pem");
var ecPublicKey  = file.readbytes("ec-public.pem");
var token = crypto.jwtCreate("{}", ecPrivateKey, "ES256");
var payload = crypto.jwtVerify(token, ecPublicKey, "ES256");

// === EdDSA (Ed25519) ===
var keys = crypto.ed25519Generate();
var token = crypto.jwtCreate('{"data":"test"}', keys.private, "EdDSA");
var payload = crypto.jwtVerify(token, keys.public, "EdDSA");

// === Additional headers ===
var token = crypto.jwtCreate(
    "{}", "secret", "HS256",
    #{"kid": "key_001", "x5c": "cert_data"}
);

// === Automatic exp check ===
var now = datetime.timestamp(datetime.now());
var token = crypto.jwtCreate(
    #{"sub": "user1", "exp": now + 3600},
    "secret"
);
print(crypto.jwtVerify(token, "secret").isnull());  // false (not expired yet)

// === Decoding without verification ===
var parts = crypto.jwtDecode(token);
print(parts.header);     // '{"alg":"HS256","typ":"JWT"}'
print(parts.payload);    // '{"sub":"user1",...}'
print(parts.signature);  // "base64url_signature"

Example: File Encryption

if (!loadPlugin("plugins/crypto.dll")) { system.exit(1); }

var key = "my_secret_key";

// Reading the file
var data = file.readbytes("document.pdf");

// Encryption
var encrypted = crypto.aesEncrypt(data, key);

// Saving the encrypted data
file.writebytes("document.pdf.enc", encrypted);
print("Encrypted: " + encrypted.size() + " bytes");

// Decryption
var loaded = file.readbytes("document.pdf.enc");
var decrypted = crypto.aesDecrypt(loaded, key);
file.writebytes("document_restored.pdf", decrypted);
print("Decrypted!");

Example: HMAC Authentication

var secret = "server_secret_key";

// Client: signing the request
var request = "amount=100&to=user2";
var sig = crypto.hmac(secret, request);

// Server: verification
var ok = crypto.hmacVerify(secret, request, sig);
if (ok) {
    print("Request is authentic!");
} else {
    print("FORGERY!");
}

Example: Secure Password Storage

var password = "user_password";

// Generating the salt
var salt = crypto.randomBytes(16);

// Key derivation
var hash = crypto.pbkdf2(password, salt, 100000, 32);

// Saving: salt + hash
print("Salt: " + crypto.hexEncode(salt));
print("Hash: " + crypto.hexEncode(hash));

// Password check
var check = crypto.pbkdf2(password, salt, 100000, 32);
print(check.toString() == hash.toString());  // 1
