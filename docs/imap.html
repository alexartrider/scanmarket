Подключение и авторизация

МетодАргументыВозвращаетОписание
connecthost, [port], [tls]int (1/0)Подключение к IMAP-серверу. port по умолчанию 143 (993 если tls=true). tls — шифрование SSL/TLS
starttls—int (1/0)Переключение на TLS (после подключения без шифрования)
loginuser, passint (1/0)Авторизация по логину и паролю
authenticateuser, tokenint (1/0)Авторизация через XOAUTH2 (Gmail, Yandex)
disconnect——Закрытие соединения
ping—int (1/0)NOOP-команда — поддержание соединения, проверка активности
getStatus—tupleСтатус подключения

Возвращаемое значение getStatus()

ПолеТипОписание
connectedint1 — подключён, 0 — нет
loggedInint1 — авторизован, 0 — нет
sslEnabledint1 — SSL активен, 0 — нет
currentFolderstringТекущая выбранная папка
var imap = ImapClient();

// Подключение к Mail.ru с SSL
var ok = imap.connect("imap.mail.ru", 993, true);
print("Подключение: " + ok);  // 1

// Авторизация
ok = imap.login("user@mail.ru", "password");
print("Вход: " + ok);  // 1

// Проверка статуса
var st = imap.getStatus();
print("Подключён: " + st.connected);     // 1
print("Авторизован: " + st.loggedIn);      // 1
print("SSL: " + st.sslEnabled);             // 1
print("Папка: " + st.currentFolder);        // (пусто)

// Поддержание соединения
imap.ping();

Поиск писем

МетодАргументыВозвращаетОписание
search[criteria]array of intПоиск UID писем по IMAP-критерию. Без аргументов — все письма

IMAP-критерии поиска

КритерийОписаниеПример
ALLВсе сообщенияsearch("ALL")
UNSEENНепрочитанныеsearch("UNSEEN")
SEENПрочитанныеsearch("SEEN")
FROM "string"От отправителяsearch("FROM \"user@example.com\"")
TO "string"Получательsearch("TO \"me@example.com\"")
SUBJECT "string"Тема письмаsearch("SUBJECT \"test\"")
BODY "string"Текст телаsearch("BODY \"important\"")
SINCE dateПосле датыsearch("SINCE 01-Jan-2024")
BEFORE dateДо датыsearch("BEFORE 31-Dec-2023")
UID min:maxДиапазон UIDsearch("UID 1:100")
Larger nБольше n байтsearch("Larger 1024")
NOT критерийОтрицаниеsearch("NOT SEEN")
OR (кр1) (кр2)Логическое ИЛИsearch("OR (FROM \"a\") (FROM \"b\")")
// Все письма
var all = imap.search("ALL");
print("Всего: " + all.size());

// Непрочитанные
var unread = imap.search("UNSEEN");
print("Непрочитанных: " + unread.size());

// Последние 10 писем
var recent = [];
var start = max(0, all.size() - 10);
for (var i = start; i < all.size(); i++) {
    recent.push(all[i]);
}

// Поиск UID и фильтрация по заголовкам в скрипте
var uids = imap.search("ALL");
for (var i = 0; i < uids.size(); i++) {
    var h = imap.header(uids[i]);
    if (!h.isnull()) {
        var subj = h["Subject"];
        if (!subj.isnull() && subj.toString().contains("Important")) {
            print("Найдено: UID=" + uids[i] + "Subject=" + subj);
        }
    }
}

Чтение писем

МетодАргументыВозвращаетОписание
headeruidtupleЗаголовки письма как кортеж key-value (From, To, Subject, Date, ...)
fetchuid, [part]stringЧтение содержимого письма
attachmentsuidarray of tupleСписок вложений из BODYSTRUCTURE
fetchattachmentuid, partNo, [path_or_bytes]int / bytearrayСкачивание вложения в файл (указать путь) или ByteArray (без пути)

Параметры part для fetch()

ЗначениеОписание
BODY.PEEK[HEADER]Только заголовки (не помечает как прочитанное)
BODY.PEEK[TEXT]Только тело (не помечает как прочитанное)
BODY.PEEK[]Заголовки + тело (не помечает как прочитанное)
BODY[HEADER]Заголовки (помечает как прочитанное)
BODY[TEXT]Тело (помечает как прочитанное)
BODY[]Заголовки + тело (помечает как прочитанное)
BODY[1]Первая часть (текстовая)
BODY[2]Вторая часть (обычно вложение)

Поля возвращаемого значения attachments()

ПолеТипОписание
partNostringНомер части (для fetchattachment)
filenamestringИмя файла
typestringMIME-тип (application, text, image...)
subtypestringПодтип (pdf, octet-stream, plain...)
sizeintРазмер в байтах
var uid = uids[0];

// --- Заголовки ---
var h = imap.header(uid);
print("От:    " + h["From"]);
print("Кому:  " + h["To"]);
print("Тема:  " + h["Subject"]);
print("Дата:  " + h["Date"]);

// --- Тело письма ---
var body = imap.fetch(uid, "BODY.PEEK[TEXT]");
print(body);

// --- Заголовки + тело (без пометки прочитанного) ---
var full = imap.fetch(uid, "BODY.PEEK[]");

// --- Вложения ---
var atts = imap.attachments(uid);
print("Вложений: " + atts.size());
for (var i = 0; i < atts.size(); i++) {
    print("  [" + i + "] " + atts[i].filename
        + " (" + atts[i].type + "/" + atts[i].subtype + ") "
        + atts[i].size + " байт");
}

// --- Скачать вложение в файл ---
if (atts.size() > 0) {
    var saved = imap.fetchattachment(uid, atts[0].partNo, "C:/temp/att.pdf");
    print("Сохранено: " + saved + " байт");
}

// --- Скачать вложение в ByteArray ---
if (atts.size() > 0) {
    var data = imap.fetchattachment(uid, atts[0].partNo);
    print("Размер: " + data.size() + " байт");
}

Удаление и управление флагами

МетодАргументыВозвращаетОписание
deleteuidint (1/0)Удаление письма: помечает \Deleted + EXPUNGE
setflaguid, flag, [value]int (1/0)Установка/снятие флага. value: true (по умолчанию) или false
copyuid, folderint (1/0)Копирование письма в другую папку
moveuid, folderint (1/0)Перемещение письма (COPY + STORE \Deleted + EXPUNGE)

Флаги IMAP

ФлагОписание
\SeenПрочитано
\AnsweredНа письмо было отвечено
\FlaggedПомечено флажком (важно)
\DeletedПомечено на удаление
\DraftЧерновик
// Удалить письмо (STORE +FLAGS \Deleted + EXPUNGE)
var ok = imap.delete(uid);
print("Удалено: " + ok);

// Пометить как прочитанное
imap.setflag(uid, "\\Seen");

// Пометить флажком "важно"
imap.setflag(uid, "\\Flagged");

// Снять флажок
imap.setflag(uid, "\\Flagged", false);

// Пометить как черновик
imap.setflag(uid, "\\Draft");

// Копировать в папку "Backup"
imap.copy(uid, "Backup");

// Переместить в архив
imap.move(uid, "Archive/2024");

Пример: получение почты

if (!loadPlugin("plugins/imap.dll")) {
    print("imap.dll не загружен!");
    system.exit(1);
}

var imap = ImapClient();
imap.connect("imap.mail.ru", 993, true);
imap.login("user@mail.ru", "password");

// Выбрать INBOX
var count = imap.select("INBOX");
print("Писем: " + count);

// Найти непрочитанные
var uids = imap.search("UNSEEN");
print("Непрочитанных: " + uids.size());

// Прочитать каждое
for (var i = 0; i < uids.size(); i++) {
    var h = imap.header(uids[i]);
    if (!h.isnull()) {
        print("--- Письмо " + uids[i] + " ---");
        print("  От: " + h["From"]);
        print("  Тема: " + h["Subject"]);
        print("  Дата: " + h["Date"]);
    }
}

imap.disconnect();

Пример: отправка и удаление

if (!loadPlugin("plugins/smtp.dll")) { system.exit(1); }
if (!loadPlugin("plugins/imap.dll")) { system.exit(1); }

// Отправка письма
var smtp = SmtpClient();
smtp.connect("smtp.yandex.com", 465, true);
smtp.login("sender@yandex.ru", "password");
smtp.send("sender@yandex.ru", "recipient@mail.ru",
    "Тест", "Привет из QodeScript!");
smtp.disconnect();

// Чтение и удаление
var imap = ImapClient();
imap.connect("imap.mail.ru", 993, true);
imap.login("recipient@mail.ru", "password");
imap.select("INBOX");

// Ищем по заголовкам
var all = imap.search("ALL");
for (var i = 0; i < all.size(); i++) {
    var h = imap.header(all[i]);
    if (!h.isnull() && h["Subject"].toString() == "Тест") {
        print("Найдено! Удаляю...");
        imap.delete(all[i]);
        print("Удалено!");
        break;
    }
}

imap.disconnect();

Обработка ошибок

var imap = ImapClient();

// Проверка подключения
if (!imap.connect("imap.mail.ru", 993, true)) {
    print("Не удалось подключиться!");
    system.exit(1);
}

// Проверка логина
if (!imap.login("user@mail.ru", "wrong")) {
    print("Неверный логин/пароль!");
    imap.disconnect();
    system.exit(1);
}

// Проверка select
var count = imap.select("INBOX");
if (count == -1) {
    print("Папка не найдена!");
}

// Проверка fetch
var body = imap.fetch(999999);
if (body.isnull()) {
    print("Письмо не найдено!");
}

// Проверка getStatus()
var st = imap.getStatus();
if (!st.connected) {
    print("Соединение потеряно!");
}

imap.disconnect();
